## Technical Notes
Model:
```php
class AnalysisHistory extends Model
{
    protected $fillable = [
        'user_id', 'site_id', 'tool_name', 'input_url',
        'status', 'results', 'error_message', 'analysis_duration_ms', 'analyzed_at'
    ];

    protected $casts = [
        'results' => 'array',
        'analyzed_at' => 'datetime',
    ];

    public function user(): BelongsTo { return $this->belongsTo(User::class); }
    public function site(): BelongsTo { return $this->belongsTo(Site::class); }
    public function costs(): HasMany { return $this->hasMany(AnalysisCost::class, 'analysis_id'); }

    // Scopes
    public function scopeCompleted($query) { return $query->where('status', 'completed'); }
    public function scopeByTool($query, string $tool) { return $query->where('tool_name', $tool); }
}
```
