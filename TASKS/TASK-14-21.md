export const auditHandlers = [
  rest.get('/api/v1/sites/:siteId/audits', (req, res, ctx) => {
    return res(ctx.json(mockAuditList(5)));
  }),
  
  rest.post('/api/v1/sites/:siteId/audits', (req, res, ctx) => {
    return res(ctx.status(201), ctx.json(mockAudit({ status: 'pending' })));
  }),
  
  rest.get('/api/v1/audits/:auditId/issues', (req, res, ctx) => {
    return res(ctx.json({ data: mockIssueList(10), meta: { total: 10 } }));
  }),
  
  rest.patch('/api/v1/issues/:issueId', (req, res, ctx) => {
    return res(ctx.json({ id: 1, status: 'fixed' }));
  }),
];
