## Description
Implement middleware or a guard component to protect routes that require authentication. Redirect unauthenticated users to login with their intended destination preserved.

## Problem
1. Protected routes accessible by URL manipulation
2. No centralized route protection
3. Users see errors instead of login prompt

## Solution
1. Create auth guard component or use Next.js middleware
2. Check authentication before rendering protected routes
3. Redirect to login with return URL stored

## Acceptance Criteria
- [ ] Protected routes check authentication
- [ ] Unauthenticated users redirected to /login
- [ ] Intended destination stored for post-login redirect
- [ ] Middleware runs before page render
- [ ] No flash of protected content
- [ ] Unit tests for route protection

### Code Changes
```typescript
// middleware.ts
export function middleware(request: NextRequest) {
  const token = request.cookies.get('seoindicator_token');
  
  if (!token && isProtectedRoute(request.nextUrl.pathname)) {
    const loginUrl = new URL('/login', request.url);
    loginUrl.searchParams.set('redirect', request.nextUrl.pathname);
    return NextResponse.redirect(loginUrl);
  }
}

// Or AuthGuard component
function AuthGuard({ children }) {
  const { isAuthenticated, isLoading } = useAuth();
  const router = useRouter();
  const pathname = usePathname();
  
  useEffect(() => {
    if (!isLoading && !isAuthenticated) {
      router.push(`/login?redirect=${encodeURIComponent(pathname)}`);
    }
  }, [isAuthenticated, isLoading]);
  
  if (isLoading) return <Loading />;
  if (!isAuthenticated) return null;
  
  return children;
}
```
