## Deliverables

### 1. Sitemap Generation (`frontend/app/sitemap.ts`)

```typescript
import { MetadataRoute } from 'next';
import { COMPARISON_SLUGS } from '@/lib/compare/registry';

export default function sitemap(): MetadataRoute.Sitemap {
  const baseUrl = 'https://seoindicator.com';
  
  // Existing sitemap entries...
  
  // Add comparison pages
  const comparisonEntries = COMPARISON_SLUGS.map((slug) => ({
    url: `${baseUrl}/compare/${slug}/`,
    lastModified: new Date(),
    changeFrequency: 'monthly' as const,
    priority: 0.8, // High priority for conversion pages
  }));
  
  // Add comparison index page
  const indexEntry = {
    url: `${baseUrl}/compare/`,
    lastModified: new Date(),
    changeFrequency: 'weekly' as const,
    priority: 0.7,
  };
  
  return [
    // ... existing entries
    indexEntry,
    ...comparisonEntries,
  ];
}
```

### 2. Sitemap Index (if needed)

If sitemap becomes too large, split into:

```typescript
// sitemap-index.ts
export default function sitemapIndex(): MetadataRoute.SitemapIndex {

```
