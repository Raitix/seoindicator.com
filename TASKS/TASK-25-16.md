## Technical Notes
```php
class AnalysisHistoryController
{
    public function index(Request $request, AnalysisHistoryRepository $repo)
    {
        $validated = $request->validate([
            'page' => 'integer|min:1',
            'perPage' => 'integer|min:1|max:100',
            'tool' => 'string|nullable',
            'status' => 'in:pending,completed,failed|nullable',
            'from' => 'date|nullable',
            'to' => 'date|nullable',
        ]);

        $analyses = $repo->findByUser(auth()->id(), $validated);

        return AnalysisResource::collection($analyses);
    }

    public function show(AnalysisHistory $analysis)
    {
        if ($analysis->user_id !== auth()->id()) {
            abort(403, 'Unauthorized access to analysis.');
        }

        return new AnalysisResource($analysis->load('costs'));
    }

    public function destroy(AnalysisHistory $analysis)
    {
        if ($analysis->user_id !== auth()->id()) {
            abort(403, 'Unauthorized access to analysis.');
        }

        $analysis->delete();

        return response()->json(['success' => true, 'message' => 'Analysis deleted.']);
    }
}
```
