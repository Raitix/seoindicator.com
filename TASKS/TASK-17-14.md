## Technical Details

### Database Schema
```sql
CREATE TABLE experiments (
    id BIGINT UNSIGNED PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(255) NOT NULL UNIQUE,
    variants JSON NOT NULL, -- ["control", "variant_a", "variant_b"]
    traffic_allocation JSON, -- {"control": 50, "variant_a": 50}
    active BOOLEAN DEFAULT true,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE experiment_assignments (
    id BIGINT UNSIGNED PRIMARY KEY AUTO_INCREMENT,
    user_id BIGINT UNSIGNED NOT NULL,
    experiment_id BIGINT UNSIGNED NOT NULL,
    variant VARCHAR(50) NOT NULL,
    assigned_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    FOREIGN KEY (experiment_id) REFERENCES experiments(id) ON DELETE CASCADE,
    UNIQUE KEY user_experiment (user_id, experiment_id)
);
```

### Assignment Logic
```php
public function assignVariant(User $user, Experiment $experiment): string
{
    // Check existing assignment
    $existing = ExperimentAssignment::where('user_id', $user->id)
        ->where('experiment_id', $experiment->id)
        ->first();
    
    if ($existing) {
        return $existing->variant;
    }
    
    // Assign new variant based on traffic allocation
    $variant = $this->selectVariant($experiment->traffic_allocation);
    
    ExperimentAssignment::create([
        'user_id' => $user->id,
        'experiment_id' => $experiment->id,
        'variant' => $variant
    ]);
    
    return $variant;
}
```
