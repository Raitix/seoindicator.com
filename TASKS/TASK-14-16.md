### API Integration
```typescript
// hooks/useAuditIssues.ts
export function useAuditIssues(auditId: string, filters: IssueFilters) {
  return useQuery({
    queryKey: ['audit-issues', auditId, filters],
    queryFn: () => fetchAuditIssues(auditId, filters),
  });
}

export function useUpdateIssueStatus() {
  return useMutation({
    mutationFn: (params: { issueId: number; status: string }) =>
      updateIssueStatus(params),
    onSuccess: () => queryClient.invalidateQueries(['audit-issues']),
  });
}

export function useBulkUpdateIssues() {
  return useMutation({
    mutationFn: (params: { auditId: string; issueIds: number[]; status: string }) =>
      bulkUpdateIssues(params),
    onSuccess: () => queryClient.invalidateQueries(['audit-issues']),
  });
}
```
