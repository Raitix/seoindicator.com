### API Integration
```typescript
// hooks/useAudits.ts
export function useAudits(siteId: string) {
  return useQuery({
    queryKey: ['audits', siteId],
    queryFn: () => fetchAudits(siteId),
    refetchInterval: (data) => {
      // Poll every 5s if any audit is running
      const hasRunning = data?.some(a => a.status === 'running');
      return hasRunning ? 5000 : false;
    },
  });
}

export function useStartAudit() {
  return useMutation({
    mutationFn: (params: StartAuditParams) => startAudit(params),
    onSuccess: () => queryClient.invalidateQueries(['audits']),
  });
}
```
