### API Client
```typescript
// frontend/lib/api/backlinks.ts
export const backlinksApi = {
  getOverview: (projectId: number) => 
    fetch(`/api/v1/projects/${projectId}/backlinks`).then(r => r.json()),
  
  getDomains: (projectId: number, params: { page: number; sort: string }) =>
    fetch(`/api/v1/projects/${projectId}/backlinks/domains?${new URLSearchParams(params)}`)
      .then(r => r.json()),
  
  getAnchors: (projectId: number) =>
    fetch(`/api/v1/projects/${projectId}/backlinks/anchors`).then(r => r.json()),
  
  getHistory: (projectId: number, from: string, to: string) =>
    fetch(`/api/v1/projects/${projectId}/backlinks/history?from=${from}&to=${to}`)
      .then(r => r.json()),
  
  triggerSync: (projectId: number) =>
    fetch(`/api/v1/projects/${projectId}/backlinks/sync`, { method: 'POST' })
      .then(r => r.json()),
};
```
