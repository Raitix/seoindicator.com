## Cache Key Strategy

```php
class AnalysisCacheService
{
    private const CACHE_PREFIX = 'ai_analysis:';
    private const DEFAULT_TTL = 86400; // 24 hours
    
    public function generateKey(string $url, string $html): string
    {
        // Normalize URL
        $normalizedUrl = $this->normalizeUrl($url);
        
        // Hash HTML content
        $htmlHash = hash('sha256', $html);
        
        // Create versioned key (v1 = unified prompt format)
        return sprintf(
            '%sv1:%s:%s',
            self::CACHE_PREFIX,
            hash('sha256', $normalizedUrl),
            substr($htmlHash, 0, 16)
        );
    }
    
    private function normalizeUrl(string $url): string
    {
        // Remove protocol, www, trailing slash
        $url = strtolower($url);
        $url = preg_replace('#^https?://#', '', $url);
        $url = preg_replace('#^www\.#', '', $url);
        $url = rtrim($url, '/');
        
        return $url;
    }
    
    public function get(string $key): ?AnalysisResult
    {
        $cached = Cache::get($key);
        
        if (!$cached) {
            return null;
        }
        
        // Verify cache version compatibility
        if (isset($cached['version']) && $cached['version'] !== 1) {
            return null;
        }
        
        return AnalysisResult::fromArray($cached['data']);
    }
    
    public function put(string $key, AnalysisResult $result, int $ttl = self::DEFAULT_TTL): void
    {
        $data = [
            'version' => 1,
            'data' => $result->toArray(),
            'cached_at' => now()->toIso8601String(),
            'model' => $result->model,
        ];
        
        Cache::put($key, $data, $ttl);
    }
}
```

## Cache Statistics

```php
class AnalysisCacheService
{
    public function getStats(): array
    {
        return [
            'total_cached' => $this->countCached(),
            'by_model' => $this->countByModel(),
            'avg_age_hours' => $this->getAverageAge(),
            'hit_rate' => $this->getHitRate(),
        ];
    }
    
    public function getHitRate(): float
    {
        // Track hits/misses via separate counters
        $hits = Cache::get('ai_analysis:stats:hits', 0);
        $misses = Cache::get('ai_analysis:stats:misses', 0);
        
        $total = $hits + $misses;
        
        return $total > 0 ? round($hits / $total * 100, 1) : 0.0;
    }
}
```
