## Technical Notes
```tsx
// app/dashboard/page.tsx
export default function DashboardPage() {
  const { data: dashboard, isLoading } = useQuery({
    queryKey: ['dashboard'],
    queryFn: () => api.get('/dashboard').then(r => r.data.data),
  });

  if (isLoading) return <DashboardSkeleton />;

  return (
    <DashboardLayout>
      <WelcomeCard user={user} />
      
      <div className="grid grid-cols-1 md:grid-cols-4 gap-4">
        <StatCard label="Total Analyses" value={dashboard.stats.totalAnalyses} />
        <StatCard label="This Month" value={dashboard.stats.thisMonth} />
        <StatCard label="This Week" value={dashboard.stats.thisWeek} />
        <StatCard label="Today" value={dashboard.stats.today} />
      </div>

      <div className="grid grid-cols-1 lg:grid-cols-3 gap-6 mt-6">
        <CostSummaryCard data={dashboard.costSummary} />
        <QuickInsightsCard data={dashboard.quickInsights} />
        <QuickActionsCard />
      </div>

      <RecentAnalysesSection analyses={dashboard.recentAnalyses} />
    </DashboardLayout>
  );
}
```
