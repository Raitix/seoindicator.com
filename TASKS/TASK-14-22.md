```typescript
import { test, expect } from '@playwright/test';

test.describe('Audit Flow', () => {
  test.beforeEach(async ({ page }) => {
    // Login and navigate to site
    await page.goto('/login');
    await page.fill('[name="email"]', process.env.TEST_USER_EMAIL);
    await page.fill('[name="password"]', process.env.TEST_USER_PASSWORD);
    await page.click('button[type="submit"]');
    await page.waitForURL('/dashboard');
  });
  
  test('complete audit flow', async ({ page }) => {
    // 1. Navigate to site
    await page.goto('/sites/1');
    
    // 2. See audit widget
    await expect(page.locator('[data-testid="audit-widget"]')).toBeVisible();
    
    // 3. Start new audit
    await page.click('button:has-text("Start New")');
    await page.fill('input[name="max_pages"]', '10');
    await page.click('button:has-text("Start Audit")');
    
    // 4. Verify audit started
    await expect(page.locator('text=Audit in progress')).toBeVisible();
    
    // 5. Wait for completion (or use mock)
    await page.waitForSelector('text=Completed', { timeout: 60000 });
    
    // 6. Navigate to audit detail
    await page.click('a:has-text("View Details")');
    await expect(page).toHaveURL(/\/sites\/\d+\/audits\/\d+/);
    
    // 7. View issues
    await expect(page.locator('[data-testid="issue-list"]')).toBeVisible();
    
    // 8. Filter issues
    await page.selectOption('select[name="severity"]', 'critical');
    await expect(page.locator('[data-testid="issue-item"]')).toHaveCount(expect.any(Number));
    
    // 9. Mark issue as fixed
    const firstIssue = page.locator('[data-testid="issue-item"]').first();
    await firstIssue.click('button:has-text("Mark Fixed")');
    await expect(firstIssue.locator('.status-badge:has-text("Fixed")')).toBeVisible();
    
    // 10. Verify in issue list
    await page.reload();
    await expect(firstIssue.locator('.status-badge:has-text("Fixed")')).toBeVisible();
  });
  
  test('audit widget shows running state', async ({ page }) => {
    // Mock running audit
    await page.route('**/api/v1/sites/*/audits*', route => {
      route.fulfill({
        json: {
          data: [{
            id: 1,
            status: 'running',
            pages_crawled: 50,
            issues_found: 12,
          }],
        },
      });
    });
    
    await page.goto('/sites/1');
    
    // Verify progress bar
    await expect(page.locator('.progress-bar')).toBeVisible();
    await expect(page.locator('text=50 pages crawled')).toBeVisible();
  });
  
  test('audit list page', async ({ page }) => {
    await page.goto('/sites/1/audits');
    
    // Verify list loads
    await expect(page.locator('[data-testid="audit-list"]')).toBeVisible();
    
    // Verify audit cards
    const auditCards = page.locator('[data-testid="audit-card"]');
    await expect(auditCards).toHaveCount(expect.any(Number));
    
    // Click to view detail
    await auditCards.first().click();
    await expect(page).toHaveURL(/\/audits\/\d+/);
  });
  
  test('bulk issue management', async ({ page }) => {
    await page.goto('/sites/1/audits/1');
    
    // Select multiple issues
    await page.check('input[name="select-all"]');
    
    // Verify all selected
    const checkboxes = page.locator('input[type="checkbox"]:checked');
    await expect(checkboxes).toHaveCount(expect.any(Number));
    
    // Bulk ignore
    await page.click('button:has-text("Ignore")');
    
    // Verify all marked ignored
    await page.waitForSelector('text=issues marked as ignored');
  });
  
  test('audit cancellation', async ({ page }) => {
    // Mock running audit
    await page.route('**/api/v1/audits/1', route => {
      route.fulfill({
        json: { id: 1, status: 'running' },
      });
    });
    
    await page.goto('/sites/1/audits');
    
    // Click cancel
    await page.click('button:has-text("Cancel")');
    
    // Confirm cancellation
    await page.click('button:has-text("Confirm")');
    
    // Verify cancelled
    await expect(page.locator('text=Cancelled')).toBeVisible();
  });
});
```

**CRITICAL FINDING: Missing Authentication in E2E Tests**

**Root Cause:** All audit detail page tests (E2E-AUD-010 through E2E-AUD-030) fail because the pages are **protected routes** that redirect unauthenticated users to the Sign In page.

**Evidence from Error Context:**
```yaml
# Page snapshot shows Sign In page instead of Audit Detail
- generic: "Sign In"
- textbox "Email"
- textbox "Password"
```

The `AuthContext.tsx` redirects to `/login` for any route not in the `publicRoutes` list. The audit routes (`/sites/:siteId/audits`) are NOT public routes.

**Auth Protection Code (AuthContext.tsx:line ~120):**
```typescript
// Redirect unauthenticated users to login for protected routes
if (!isAuthenticated && !isPublicRoute(pathname)) {
  storeRedirectUrl(pathname);
  router.push("/login");
}
```

**Why Some Tests Pass:**
- E2E-AUD-001/002/004: These tests check for elements that may appear briefly before the redirect completes, or they check for generic elements that also exist on the login page
- E2E-AUD-019: Checks for H1 which exists on the login page ("Sign In")
- E2E-AUD-029: Only checks that page doesn't crash during timeout

**Required Fix for E2E Tests:**

Tests must mock authentication. Options:

1. **Add test authentication in beforeEach:**
```typescript
test.beforeEach(async ({ page }) => {
  // Mock authenticated session
  await page.addInitScript(() => {
    localStorage.setItem('seoindicator_user', JSON.stringify({
      id: 1,
      email: 'test@example.com',
      name: 'Test User',
      role: 'user'
    }));
    localStorage.setItem('seoindicator_token', 'test-token');
  });
  // Rest of setup...
});
```

2. **Or use Playwright storageState for authenticated tests:**
```typescript
// playwright.config.ts
export default defineConfig({
  use: {
    storageState: 'playwright/.auth/user.json',
  },
});
```

3. **Or create a test-specific auth bypass (not recommended for production):**
```typescript
// In AuthContext, check for test environment
if (process.env.PLAYWRIGHT_TEST === 'true' && !isAuthenticated) {
  // Auto-authenticate with test user
}
```

**Recommended Approach:** Option 1 (localStorage mock) is simplest and doesn't require additional test infrastructure.
