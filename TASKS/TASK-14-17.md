```typescript
interface AuditWidgetProps {
  siteId: string;
  lastAudit?: {
    id: number;
    status: string;
    issues: {
      critical: number;
      warning: number;
      notice: number;
    };
    completed_at: string;
  };
  runningAudit?: {
    id: number;
    progress: number;
    pages_crawled: number;
    issues_found: number;
  };
  recentIssues?: Array<{
    id: number;
    description: string;
    severity: string;
  }>;
  onStartAudit: () => void;
  onViewAll: () => void;
}
```
