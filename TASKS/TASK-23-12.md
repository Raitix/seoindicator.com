### Service Implementation
```php
class BacklinkSuggestionService
{
    /**
     * Generate suggestions for a project
     */
    public function generate(Project $project): array
    {
        $metrics = $this->getMetrics($project);
        $suggestions = [];
        
        if ($this->needsMoreDomains($metrics)) {
            $suggestions[] = $this->createSuggestion(
                'low_domains',
                'Build more referring domains. Guest posting and content marketing can help.',
                'high',
                $metrics
            );
        }
        
        if ($this->hasHighExactMatchAnchors($metrics)) {
            $suggestions[] = $this->createSuggestion(
                'high_exact_match',
                'Diversify anchor text. Too many exact-match anchors can appear manipulative.',
                'high',
                $metrics
            );
        }
        
        // ... more rules
        
        return $this->prioritize($suggestions);
    }
}
```
