```tsx
// app/dashboard/history/page.tsx
export default function HistoryPage() {
  const [filters, setFilters] = useState({ tool: '', status: '', from: '', to: '' });
  const [page, setPage] = useState(1);

  const { data, isLoading } = useQuery({
    queryKey: ['analyses', { page, ...filters }],
    queryFn: () => api.get('/analyses', { params: { page, perPage: 15, ...filters } })
      .then(r => r.data),
  });

  return (
    <DashboardLayout>
      <Card>
        <CardHeader>
          <CardTitle>Analysis History</CardTitle>
          <HistoryFilters filters={filters} onChange={setFilters} />
        </CardHeader>
        <CardContent>
          {isLoading ? (
            <HistorySkeleton />
          ) : data.data.length === 0 ? (
            <EmptyState message="No analyses found" />
          ) : (
            <>
              <HistoryTable analyses={data.data} />
              <Pagination meta={data.meta} onChange={setPage} />
            </>
          )}
        </CardContent>
      </Card>
    </DashboardLayout>
  );
}
```
