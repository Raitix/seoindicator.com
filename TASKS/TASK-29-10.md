import { render, screen, fireEvent, waitFor } from '@testing-library/react';
import { ExportResultsButton } from '../ExportResultsButton';

describe('ExportResultsButton', () => {
  const mockData = {
    analyzedAt: '2026-02-28T12:00:00Z',
    results: { title: { value: 'Test', status: 'ok' } },
  };
  
  it('renders button with correct text', () => {
    render(
      <ExportResultsButton 
        toolName="test-tool" 
        url="https://example.com" 
        data={mockData} 
      />
    );
    
    expect(screen.getByText('Export Results')).toBeInTheDocument();
  });
  
  it('button disabled when no results', () => {
    render(
      <ExportResultsButton 
        toolName="test-tool" 
        url="https://example.com" 
        data={null} 
      />
    );
    
    expect(screen.getByRole('button')).toBeDisabled();
  });
  
  it('opens modal on click', () => {
    render(
      <ExportResultsButton 
        toolName="test-tool" 
        url="https://example.com" 
        data={mockData} 
      />
    );
    
    fireEvent.click(screen.getByText('Export Results'));
    
    expect(screen.getByText('Export Results')).toBeInTheDocument();
    expect(screen.getByLabelText('PDF')).toBeInTheDocument();
    expect(screen.getByLabelText('XLS')).toBeInTheDocument();
  });
});
