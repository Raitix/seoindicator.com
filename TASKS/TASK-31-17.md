## Performance Benchmarks

### Benchmark Scenarios

```php
class UnifiedVsLegacyAnalysisBenchmark
{
    private const TEST_PAGES = [
        'small' => 'fixtures/small_page.html',      // ~10KB
        'medium' => 'fixtures/medium_page.html',    // ~100KB
        'large' => 'fixtures/large_page.html',      // ~500KB
    ];
    
    /**
     * @benchmark
     * @iterations 10
     */
    public function benchmark_unified_small_page(): array
    {
        return $this->measureAnalysis('unified', 'small');
    }
    
    /**
     * @benchmark
     * @iterations 10
     */
    public function benchmark_legacy_small_page(): array
    {
        return $this->measureAnalysis('legacy', 'small');
    }
    
    // ... similar for medium and large
    
    private function measureAnalysis(string $mode, string $size): array
    {
        $html = file_get_contents(self::TEST_PAGES[$size]);
        
        $start = microtime(true);
        $startMemory = memory_get_usage();
        
        if ($mode === 'unified') {
            $result = $this->aiAnalysisService->analyze($url, $html);
        } else {
            $result = $this->legacyService->run($url);
        }
        
        $endMemory = memory_get_usage();
        $end = microtime(true);
        
        return [
            'duration_ms' => ($end - $start) * 1000,
            'memory_bytes' => $endMemory - $startMemory,
            'api_calls' => $mode === 'unified' ? 1 : 18,
            'tokens_used' => $result->totalTokens,
        ];
    }
}
```
