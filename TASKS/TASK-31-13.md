## Updated Implementation

```php
<?php

namespace App\Services\Analysis;

use App\Models\SEOAnalysis;
use App\Models\AIAnalysisRequest;
use App\Services\AI\AIAnalysisService;
use App\Services\AI\DTOs\AnalysisResult;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Log;

class AnalysisRunnerService
{
    public function __construct(
        private AIAnalysisService $aiAnalysisService,
        private HtmlFetcherService $htmlFetcher,
        private AnalysisRepository $repository,
    ) {}
    
    public function run(string $url, array $options = []): SEOAnalysis
    {
        $startTime = microtime(true);
        
        try {
            // 1. Fetch HTML content
            $html = $this->htmlFetcher->fetch($url);
            
            // 2. Execute unified AI analysis
            $result = $this->aiAnalysisService->analyze($url, $html, $options);
            
            // 3. Store analysis with metadata
            $analysis = $this->storeAnalysis($url, $html, $result);
            
            // 4. Log success
            Log::info('Unified AI analysis completed', [
                'url' => $url,
                'model' => $result->model,
                'tokens' => $result->promptTokens + $result->responseTokens,
                'latency_ms' => $result->latencyMs,
            ]);
            
            return $analysis;
            
        } catch (AIAnalysisException $e) {
            Log::error('AI analysis failed', [
                'url' => $url,
                'error' => $e->getMessage(),
            ]);
            
            // Store failed analysis for retry
            return $this->storeFailedAnalysis($url, $e);
        }
    }
    
    private function storeAnalysis(string $url, string $html, AnalysisResult $result): SEOAnalysis
    {
        return DB::transaction(function () use ($url, $html, $result) {
            // Create analysis record
            $analysis = SEOAnalysis::create([
                'url' => $url,
                'domain' => parse_url($url, PHP_URL_HOST),
                'overall_score' => $result->overallScore,
                'overall_summary' => $result->overallSummary,
                'ai_model' => $result->model,
                'prompt_tokens' => $result->promptTokens,
                'response_tokens' => $result->responseTokens,
                'total_tokens' => $result->promptTokens + $result->responseTokens,
                'truncation_applied' => $result->truncationStrategies,
                'category_scores' => $result->categoryScores,
                'analysis_data' => $this->formatIndicatorData($result->indicators),
            ]);
            
            // Create AI request record
            AIAnalysisRequest::create([
                'analysis_id' => $analysis->id,
                'model' => $result->model,
                'prompt_tokens' => $result->promptTokens,
                'response_tokens' => $result->responseTokens,
                'total_tokens' => $result->promptTokens + $result->responseTokens,
                'latency_ms' => $result->latencyMs,
                'truncation_strategies' => $result->truncationStrategies,
                'status' => 'completed',
            ]);
            
            // Store individual indicator results
            $this->storeIndicatorResults($analysis, $result->indicators);
            
            return $analysis;
        });
    }
    
    private function formatIndicatorData(array $indicators): array
    {
        $data = [];
        
        foreach ($indicators as $indicator) {
            $data[$indicator['id']] = [
                'score' => $indicator['score'],
                'passed' => $indicator['passed'],
                'details' => $indicator['details'] ?? [],
                'suggestions' => $indicator['suggestions'] ?? [],
            ];
        }
        
        return $data;
    }
}
```

## Feature Flag Integration

```php
// Support gradual rollout
public function run(string $url, array $options = []): SEOAnalysis
{
    $useUnifiedPrompt = config('ai.unified_prompt_enabled', true);
    
    if (!$useUnifiedPrompt) {
        // Fall back to old multi-call approach
        return $this->runLegacy($url, $options);
    }
    
    return $this->runUnified($url, $options);
}
```
