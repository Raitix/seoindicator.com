## Technical Details

### Component Implementation
```tsx
import { LineChart, Line, XAxis, YAxis, Tooltip, Legend, ResponsiveContainer } from 'recharts';

interface BacklinkHistoryChartProps {
  projectId: number;
  dateRange?: { from: string; to: string };
}

export function BacklinkHistoryChart({ projectId, dateRange }: BacklinkHistoryChartProps) {
  const { data, isLoading } = useQuery({
    queryKey: ['backlinks-history', projectId, dateRange],
    queryFn: () => backlinksApi.getHistory(projectId, dateRange?.from, dateRange?.to),
  });

  if (isLoading) return <Skeleton className="h-64" />;
  if (!data?.data?.length) return <EmptyState message="No backlink history yet" />;

  return (
    <Card>
      <CardHeader>
        <div className="flex justify-between items-center">
          <CardTitle>Backlink History</CardTitle>
          <DateRangeSelector value={dateRange} onChange={setDateRange} />
        </div>
      </CardHeader>
      <CardContent>
        <ResponsiveContainer width="100%" height={300}>
          <LineChart data={data.data}>
            <XAxis dataKey="date" />
            <YAxis yAxisId="left" />
            <YAxis yAxisId="right" orientation="right" />
            <Tooltip content={<CustomTooltip />} />
            <Legend />
            <Line yAxisId="left" type="monotone" dataKey="total_backlinks" stroke="#2563eb" name="Backlinks" />
            <Line yAxisId="left" type="monotone" dataKey="referring_domains_count" stroke="#16a34a" name="Domains" />
            <Line yAxisId="right" type="monotone" dataKey="avg_domain_trust" stroke="#f59e0b" name="Trust Score" />
          </LineChart>
        </ResponsiveContainer>
      </CardContent>
    </Card>
  );
}
```
