## Controller Implementation

```php
<?php

namespace App\Http\Controllers\Api\V1;

use App\Http\Controllers\Controller;
use App\Services\AI\ModelSelectionService;
use Illuminate\Http\JsonResponse;

class AIModelController extends Controller
{
    public function __construct(
        private ModelSelectionService $modelService,
    ) {}
    
    public function index(): JsonResponse
    {
        $models = $this->modelService->getAvailableModels();
        $default = $this->modelService->getDefaultModel();
        $userPreference = auth()->check()
            ? $this->modelService->getUserPreference(auth()->user())
            : null;
        
        return response()->json([
            'success' => true,
            'data' => [
                'models' => $models->map(fn ($model) => $this->formatModel($model)),
                'default_model' => $default,
                'user_preference' => $userPreference,
            ],
        ]);
    }
    
    private function formatModel(AIModel $model): array
    {
        return [
            'id' => $model->model_id,
            'name' => $model->display_name,
            'provider' => $model->provider,
            'context_window' => $model->context_window,
            'is_free' => $model->is_free_tier,
            'is_recommended' => $model->model_id === 'google/gemini-2.0-flash-exp:free',
            'capabilities' => $model->capabilities ?? ['text', 'json'],
            'avg_latency_ms' => $model->avg_latency_ms,
            'best_for' => $this->getBestForText($model),
        ];
    }
    
    private function getBestForText(AIModel $model): string
    {
        return match ($model->model_id) {
            'google/gemini-2.0-flash-exp:free' => 'Large pages (100K+ characters)',
            'qwen/qwen-2-7b-instruct:free', 'mistralai/mistral-7b-instruct:free' => 'Medium pages',
            'meta-llama/llama-3-8b-instruct:free' => 'Small pages only',
            default => 'General use',
        };
    }
}
```
