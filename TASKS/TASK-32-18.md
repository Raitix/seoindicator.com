### 2. A/B Test Hook (`frontend/hooks/useABTest.ts`)

```typescript
import { useState, useEffect } from 'react';

export function useABTest(testId: string): string {
  const [variant, setVariant] = useState<string>('a');

  useEffect(() => {
    // Check if user already assigned
    const existingVariant = localStorage.getItem(`ab-${testId}`);
    if (existingVariant) {
      setVariant(existingVariant);
      return;
    }

    // Assign variant based on weights
    const test = CTA_TESTS.find((t) => t.id === testId);
    if (!test) return;

    const random = Math.random() * 100;
    let cumulative = 0;
    for (const v of test.variants) {
      cumulative += v.weight;
      if (random < cumulative) {
        setVariant(v.id);
        localStorage.setItem(`ab-${testId}`, v.id);
        break;
      }
    }
  }, [testId]);

  return variant;
}
```

### 3. CTA Component with A/B Test Support

```typescript
export function ComparisonCTA({ variant = 'primary', abTestGroup }: ComparisonCTAProps) {
  const ctaVariant = useABTest('comparison-cta-text');
  const testConfig = getTestConfig('comparison-cta-text', ctaVariant);

  const handleClick = () => {
    // Track conversion
    analytics.track('cta_click', {
      test_id: 'comparison-cta-text',
      variant: ctaVariant,
      page: window.location.pathname,
    });
  };

  return (
    <Button onClick={handleClick}>
      {testConfig.text}
    </Button>
  );
}
```
