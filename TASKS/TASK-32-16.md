#### d. Related Comparisons
Add section at bottom of each comparison page:

```typescript
function RelatedComparisons({ slugs }: { slugs: string[] }) {
  return (
    <section>
      <h2>Related Comparisons</h2>
      <div className="grid">
        {slugs.map((slug) => (
          <Link key={slug} href={`/compare/${slug}`}>
            {formatComparisonTitle(slug)}
          </Link>
        ))}
      </div>
    </section>
  );
}
```
