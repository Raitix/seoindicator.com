import { ImageResponse } from 'next/og';

export async function GET(request: Request) {

  );
}

### 2. Generate OG Images in Metadata

```typescript
// In comparison page
export async function generateMetadata({ params }): Promise<Metadata> {
  const comparison = getComparisonBySlug(params.slug);
  const [a, b] = comparison.competitors;
  
  const ogImageUrl = `${baseUrl}/api/og?a=${encodeURIComponent(a)}&b=${encodeURIComponent(b)}`;

  return {
    title: comparison.title,
    description: comparison.metaDescription,
    openGraph: {
      title: comparison.title,
      description: comparison.metaDescription,
      images: [{ url: ogImageUrl, width: 1200, height: 630 }],
    },
    twitter: {
      card: 'summary_large_image',
      title: comparison.title,
      description: comparison.metaDescription,
      images: [ogImageUrl],
    },
  };
}
```

### [2026-03-07 17:57 GMT+2] Agent: LocalTester
**❌ LOCAL TESTS FAILED - Critical Bug Found**

**Bug Summary:** `ImageResponse` is incompatible with `output: 'export'` (static site generation)

**Root Cause:**
The Next.js project configuration (`next.config.ts`) has `output: 'export'` which enables static site generation. The `ImageResponse` class from `next/og` requires server-side rendering and is **not compatible with static export**.

**Error Message:**
```
export const dynamic = "force-static"/export const revalidate not configured on route "/api/og" with "output: export"
```

**Actual Server Response:** HTTP 500 Internal Server Error

**Why Unit Tests Passed:**
The test suite mocks `ImageResponse` and doesn't test the actual Next.js runtime compatibility. The mock extends `Response` and simulates the behavior, but the real `ImageResponse` cannot be statically exported.

**Solutions (Choose One):**

| Option | Description | Effort |
|--------|-------------|--------|
| **A. Pre-generate static images** | Create build script to generate all OG images as static files to `/public/og/` | Medium |
| **B. Use external OG service** | Use Vercel OG Image API or similar (free tier available) | Small |
| **C. Remove static export** | Change `output: 'export'` to use server-side rendering | Large (affects entire project) |
| **D. Edge Runtime with ISR** | Add `export const runtime = 'edge'` and remove static export | Medium |

**Recommended Fix:** Option A (Pre-generate static images)
- Create `scripts/generate-og-images.ts` to pre-generate all comparison OG images at build time
- Store in `/public/og/compare/{slug}.png`
- Update metadata to reference static files instead of API route
- Keeps project as static export (no server required)

**Test Evidence:**
- Unit tests: 67/67 passing (mock-based)
- Integration test: HTTP 500 on actual API route
- Root cause confirmed via Next.js error message
