## Technical Notes
```php
class DashboardController
{
    public function index(DashboardAggregator $aggregator)
    {
        return response()->json([
            'success' => true,
            'data' => $aggregator->getDashboardData(auth()->id()),
        ]);
    }

    public function stats(DashboardAggregator $aggregator)
    {
        return response()->json([
            'success' => true,
            'data' => $aggregator->getStats(auth()->id()),
        ]);
    }
}

class DashboardAggregator
{
    public function getDashboardData(int $userId): array
    {
        return Cache::remember("dashboard.{$userId}", 60, function () use ($userId) {
            return [
                'stats' => $this->getStats($userId),
                'costSummary' => $this->getCostSummary($userId),
                'recentAnalyses' => $this->getRecentAnalyses($userId),
                'quickInsights' => $this->getQuickInsights($userId),
            ];
        });
    }

    protected function getStats(int $userId): array
    {
        return [
            'totalAnalyses' => AnalysisHistory::where('user_id', $userId)->count(),
            'thisMonth' => AnalysisHistory::where('user_id', $userId)
                ->whereMonth('analyzed_at', now()->month)->count(),
            'thisWeek' => AnalysisHistory::where('user_id', $userId)
                ->whereBetween('analyzed_at', [now()->startOfWeek(), now()])->count(),
            'today' => AnalysisHistory::where('user_id', $userId)
                ->whereDate('analyzed_at', today())->count(),
        ];
    }
}
```
