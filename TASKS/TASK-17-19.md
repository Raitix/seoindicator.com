### Classification Logic
```php
class IndustryClassifier
{
    public function classify(string $url, string $html): string
    {
        // Check for e-commerce indicators (cart, checkout, products)
        if ($this->hasEcommerceSignals($html)) return 'ecommerce';
        
        // Check for SaaS indicators (pricing, features, demo)
        if ($this->hasSaaSSignals($html)) return 'saas';
        
        // Check for local business (address, phone, hours)
        if ($this->hasLocalSignals($html)) return 'local';
        
        // Default
        return 'other';
    }
}
```
