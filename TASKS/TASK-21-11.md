## Technical Notes

### API Client
```typescript
// frontend/lib/api/ai-suggestions.ts
export async function getAISuggestion(
  indicator: string,
  context: Record<string, unknown>,
  options?: { provider?: string; debug?: boolean }
): Promise<SuggestionResult> {
  const response = await fetch('/api/v1/ai/suggestions', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ indicator, context, ...options }),
  });
  
  if (!response.ok) {
    throw new Error('Failed to get AI suggestion');
  }
  
  return response.json();
}
```

### Types
```typescript
// frontend/lib/types/ai-suggestions.ts
interface Suggestion {
  type: 'improve' | 'rewrite';
  text: string;
  reason: string;
}

interface SuggestionResult {
  indicator: string;
  score: number;
  issues: string[];
  suggestions: Suggestion[];
  best_practices: string[];
  provider: string;
  model: string | null;
  generated_at: string;
  fallback_reason?: string;
}
```

### Component Usage
```tsx
<AISuggestionButton
  indicator="title"
  context={{
    title: pageTitle,
    content_summary: pageContent,
    keyword: targetKeyword,
  }}
  onSuggestion={(result) => setSuggestions(result)}
/>
```
