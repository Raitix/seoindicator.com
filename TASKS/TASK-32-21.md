```typescript
import { test, expect } from '@playwright/test';

test.describe('Comparison Pages', () => {
  test.describe('Comparison Index Page', () => {
    test('loads /compare successfully', async ({ page }) => {
      await page.goto('/compare');
      await expect(page).toHaveTitle(/Compare SEO Tools/);
    });

    test('displays Tier 1 comparisons', async ({ page }) => {
      await page.goto('/compare');
      await expect(page.getByText('Ahrefs')).toBeVisible();
      await expect(page.getByText('SEMrush')).toBeVisible();
    });

    test('links to individual comparisons work', async ({ page }) => {
      await page.goto('/compare');
      await page.click('text=vs Ahrefs');
      await expect(page).toHaveURL(/compare\/seoindicator-vs-ahrefs/);
    });
  });

  test.describe('Individual Comparison Page', () => {
    test('loads comparison page with correct title', async ({ page }) => {
      await page.goto('/compare/seoindicator-vs-ahrefs');
      await expect(page).toHaveTitle(/Ahrefs/);
    });

    test('renders feature comparison table', async ({ page }) => {
      await page.goto('/compare/seoindicator-vs-ahrefs');
      const table = page.locator('table');
      await expect(table).toBeVisible();
    });

    test('renders pricing comparison', async ({ page }) => {
      await page.goto('/compare/seoindicator-vs-ahrefs');
      await expect(page.getByText(/Pricing/i)).toBeVisible();
    });

    test('renders pros and cons cards', async ({ page }) => {
      await page.goto('/compare/seoindicator-vs-ahrefs');
      await expect(page.getByText(/Pros/i)).toBeVisible();
      await expect(page.getByText(/Cons/i)).toBeVisible();
    });

    test('CTA buttons are clickable', async ({ page }) => {
      await page.goto('/compare/seoindicator-vs-ahrefs');
      const cta = page.getByRole('link', { name: /Try.*Free/i });
      await expect(cta).toBeEnabled();
    });

    test('mobile responsive layout', async ({ page }) => {
      await page.setViewportSize({ width: 375, height: 667 });
      await page.goto('/compare/seoindicator-vs-ahrefs');
      // Verify no horizontal scroll
      const scrollWidth = await page.evaluate(() => document.body.scrollWidth);
      expect(scrollWidth).toBeLessThanOrEqual(375);
    });

    test('sticky CTA appears on mobile scroll', async ({ page }) => {
      await page.setViewportSize({ width: 375, height: 667 });
      await page.goto('/compare/seoindicator-vs-ahrefs');
      await page.mouse.wheel(0, 500);
      // Check for sticky CTA
      const stickyCta = page.locator('.sticky-cta');
      await expect(stickyCta).toBeVisible({ timeout: 5000 });
    });
  });

  test.describe('SEO Verification', () => {
    test('valid JSON-LD schema', async ({ page }) => {
      await page.goto('/compare/seoindicator-vs-ahrefs');
      const schemaScripts = await page.locator('script[type="application/ld+json"]').all();
      expect(schemaScripts.length).toBeGreaterThan(0);
      
      for (const script of schemaScripts) {
        const content = await script.textContent();
        const json = JSON.parse(content || '{}');
        expect(json['@context']).toBe('https://schema.org');
      }
    });

    test('correct meta tags', async ({ page }) => {
      await page.goto('/compare/seoindicator-vs-ahrefs');
      const title = await page.title();
      expect(title).toContain('Ahrefs');
      
      const description = await page.locator('meta[name="description"]').getAttribute('content');
      expect(description).toBeTruthy();
    });
  });

  test.describe('Navigation', () => {
    test('breadcrumb navigation works', async ({ page }) => {
      await page.goto('/compare/seoindicator-vs-ahrefs');
      await page.click('text=Compare');
      await expect(page).toHaveURL('/compare');
    });

    test('related comparisons link correctly', async ({ page }) => {
      await page.goto('/compare/seoindicator-vs-ahrefs');
      const relatedLink = page.getByRole('link', { name: /vs SEMrush/i });
      if (await relatedLink.isVisible()) {
        await relatedLink.click();
        await expect(page).toHaveURL(/semrush/);
      }
    });
  });
});
```
