## Component Structure

```tsx
// ModelPicker.tsx
'use client';

import { useAIModels } from '@/hooks/useAIModels';
import { Check, Zap, Database, Clock } from 'lucide-react';

interface ModelPickerProps {
  selectedModel: string | null;
  onModelChange: (modelId: string) => void;
  estimatedTokens?: number;
  disabled?: boolean;
}

export function ModelPicker({
  selectedModel,
  onModelChange,
  estimatedTokens,
  disabled = false,
}: ModelPickerProps) {
  const { models, defaultModel, isLoading, error } = useAIModels();
  
  if (isLoading) {
    return <ModelPickerSkeleton />;
  }
  
  if (error) {
    return <ModelPickerError error={error} />;
  }
  
  return (
    <div className="space-y-3">
      <label className="text-sm font-medium text-gray-700">
        AI Model
      </label>
      
      <div className="grid gap-2">
        {models.map((model) => {
          const isSelected = selectedModel === model.id || 
            (!selectedModel && model.id === defaultModel);
          const isRecommended = model.is_recommended;
          const fitsContent = estimatedTokens 
            ? estimatedTokens < model.context_window 
            : true;
          
          return (
            <button
              key={model.id}
              type="button"
              onClick={() => onModelChange(model.id)}
              disabled={disabled || !fitsContent}
              className={`
                relative flex items-start p-3 rounded-lg border transition-all
                ${isSelected 
                  ? 'border-blue-500 bg-blue-50' 
                  : 'border-gray-200 hover:border-gray-300'}
                ${!fitsContent ? 'opacity-50 cursor-not-allowed' : 'cursor-pointer'}
              `}
            >
              {isSelected && (
                <div className="absolute top-2 right-2">
                  <Check className="w-4 h-4 text-blue-500" />
                </div>
              )}
              
              {isRecommended && (
                <span className="absolute -top-2 -right-2 px-2 py-0.5 bg-blue-500 text-white text-xs rounded-full">
                  Recommended
                </span>
              )}
              
              <div className="flex-1">
                <div className="font-medium">{model.name}</div>
                <div className="text-sm text-gray-500">{model.provider}</div>
                
                <div className="flex gap-4 mt-2 text-xs text-gray-500">
                  <span className="flex items-center gap-1">
                    <Database className="w-3 h-3" />
                    {formatContextWindow(model.context_window)}
                  </span>
                  <span className="flex items-center gap-1">
                    <Clock className="w-3 h-3" />
                    ~{model.avg_latency_ms / 1000}s
                  </span>
                </div>
                
                {!fitsContent && (
                  <div className="mt-1 text-xs text-amber-600">
                    Content too large for this model
                  </div>
                )}
              </div>
            </button>
          );
        })}
      </div>
      
      <p className="text-xs text-gray-500">
        Model is automatically selected based on page size. Override for specific needs.
      </p>
    </div>
  );
}
```

## Types

```typescript
// types/ai-models.ts
export interface AIModel {
  id: string;
  name: string;
  provider: string;
  context_window: number;
  is_free: boolean;
  is_recommended: boolean;
  capabilities: string[];
  avg_latency_ms: number;
  best_for: string;
}

export interface AIModelsResponse {
  success: boolean;
  data: {
    models: AIModel[];
    default_model: string;
    user_preference: string | null;
  };
}
```

## API Hook

```typescript
// hooks/useAIModels.ts
import useSWR from 'swr';
import type { AIModel, AIModelsResponse } from '@/types/ai-models';

export function useAIModels() {
  const { data, error, isLoading } = useSWR<AIModelsResponse>(
    '/api/v1/ai/models',
    { revalidateOnFocus: false }
  );
  
  return {
    models: data?.data.models ?? [],
    defaultModel: data?.data.default_model ?? 'google/gemini-2.0-flash-exp:free',
    userPreference: data?.data.user_preference,
    isLoading,
    error,
  };
}
```
