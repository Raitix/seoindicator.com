### 2. Integration Tests

#### 2.1 Deploy Endpoint Rate Limiting Test
**File:** `tests/Integration/Security/DeployRateLimitIntegrationTest.php`

```php
<?php

namespace Tests\Integration\Security;

use Tests\TestCase;
use Illuminate\Support\Facades\Cache;

class DeployRateLimitIntegrationTest extends TestCase
{
    protected function setUp(): void
    {
        parent::setUp();
        Cache::flush();
    }

    /**
     * @test
     */
    public function deploy_migrate_endpoint_has_rate_limiting(): void
    {
        // First request with invalid key should get 403 (auth failure)
        $response = $this->postJson('/api/deploy/migrate', ['key' => 'invalid']);
        
        $response->assertStatus(403);
        
        // Check for rate limit headers
        $this->assertTrue(
            $response->headers->has('X-RateLimit-Limit') ||
            $response->headers->has('Retry-After') ||
            str_contains($response->getContent(), 'rate') ||
            str_contains($response->getContent(), 'Invalid deploy key'),
            'Response should indicate rate limiting or auth failure'
        );
    }

    /**
     * @test
     */
    public function deploy_rollback_endpoint_has_rate_limiting(): void
    {
        $response = $this->postJson('/api/deploy/rollback', ['key' => 'invalid']);
        
        $response->assertStatus(403);
    }

    /**
     * @test
     */
    public function rate_limit_blocks_after_threshold(): void
    {
        $validKey = config('app.deploy_key', env('DEPLOY_KEY', 'test-key'));
        
        // Make requests up to the limit
        for ($i = 0; $i < 10; $i++) {
            $response = $this->postJson('/api/deploy/migrate', ['key' => 'wrong-key']);
            
            // Should get either 403 (auth failure) or 429 (rate limited)
            $this->assertContains(
                $response->status(),
                [403, 429],
                "Request $i: Expected 403 (auth) or 429 (rate limited), got {$response->status()}"
            );
        }
    }

    /**
     * @test
     */
    public function health_endpoint_is_not_rate_limited(): void
    {
        // Health check should be accessible without rate limiting
        for ($i = 0; $i < 20; $i++) {
            $response = $this->getJson('/api/health');
            
            $response->assertStatus(200);
        }
    }

    /**
     * @test
     */
    public function rate_limit_includes_retry_after_header(): void
    {
        // Exhaust rate limit
        for ($i = 0; $i < 10; $i++) {
            $this->postJson('/api/deploy/migrate', ['key' => 'invalid']);
        }
        
        // Next request should be rate limited
        $response = $this->postJson('/api/deploy/migrate', ['key' => 'invalid']);
        
        if ($response->status() === 429) {
            $this->assertTrue(
                $response->headers->has('Retry-After') ||
                $response->headers->has('X-RateLimit-Reset'),
                'Rate limited response should include Retry-After or X-RateLimit-Reset header'
            );
        }
    }
}
```

### 3. Security Tests (Adversarial)

#### 3.1 Rate Limit Bypass Tests
**File:** `tests/Security/DeployRateLimitBypassTest.php`

```php
<?php

namespace Tests\Security;

use Tests\TestCase;
use Illuminate\Support\Facades\Cache;

class DeployRateLimitBypassTest extends TestCase
{
    protected function setUp(): void
    {
        parent::setUp();
        Cache::flush();
    }

    /**
     * @test
     */
    public function rate_limit_cannot_be_bypassed_with_different_ips(): void
    {
        // Simulate requests from different IPs
        $ips = ['192.168.1.1', '192.168.1.2', '192.168.1.3', '192.168.1.4', '192.168.1.5'];
        
        foreach ($ips as $ip) {
            $response = $this->postJson('/api/deploy/migrate', ['key' => 'invalid'], [
                'REMOTE_ADDR' => $ip,
            ]);
            
            // All should get 403 (auth failure) - rate limit should apply per IP or globally
            $this->assertContains($response->status(), [403, 429]);
        }
    }

    /**
     * @test
     */
    public function rate_limit_cannot_be_bypassed_with_header_injection(): void
    {
        $maliciousHeaders = [
            'X-Forwarded-For: 127.0.0.1',
            'X-Real-IP: 127.0.0.1',
            'X-Originating-IP: 127.0.0.1',
        ];

        foreach ($maliciousHeaders as $header) {
            [$name, $value] = explode(': ', $header);
            
            $response = $this->postJson('/api/deploy/migrate', ['key' => 'invalid'], [
                $name => $value,
            ]);
            
            // Should still be rate limited or auth-failed
            $this->assertContains($response->status(), [403, 429]);
        }
    }

    /**
     * @test
     */
    public function rate_limit_applies_to_both_endpoints(): void
    {
        // Exhaust limit on migrate endpoint
        for ($i = 0; $i < 10; $i++) {
            $this->postJson('/api/deploy/migrate', ['key' => 'invalid']);
        }
        
        // Rollback endpoint should also be rate limited (if shared limiter)
        $response = $this->postJson('/api/deploy/rollback', ['key' => 'invalid']);
        
        // Either still gets auth failure or rate limited
        $this->assertContains($response->status(), [403, 429]);
    }

    /**
     * @test
     */
    public function valid_key_still_respects_rate_limit(): void
    {
        $validKey = env('DEPLOY_KEY', 'test-deploy-key');
        
        // Make many requests even with valid key
        for ($i = 0; $i < 20; $i++) {
            $this->postJson('/api/deploy/migrate', ['key' => $validKey]);
        }
        
        // Should eventually be rate limited
        $response = $this->postJson('/api/deploy/migrate', ['key' => $validKey]);
        
        // This is intentional - even valid requests should be rate limited
        // to prevent abuse if key is compromised
        $this->assertContains($response->status(), [200, 429, 500]);
    }
}
```

### 4. Failed Auth Logging Tests

#### 4.1 Failed Authentication Logging Test
**File:** `tests/Integration/Security/DeployAuthLoggingTest.php`

```php
<?php

namespace Tests\Integration\Security;

use Tests\TestCase;
use Illuminate\Support\Facades\Log;
use Illuminate\Support\Facades\File;

class DeployAuthLoggingTest extends TestCase
{
    /**
     * @test
     */
    public function failed_auth_is_logged(): void
    {
        Log::shouldReceive('warning')
            ->atLeast()->once()
            ->withArgs(function ($message, $context) {
                return str_contains($message, 'deploy') ||
                       str_contains($message, 'auth') ||
                       str_contains($message, 'failed') ||
                       (isset($context['ip']) || isset($context['IP']));
            });
        
        $this->postJson('/api/deploy/migrate', ['key' => 'invalid-key']);
    }

    /**
     * @test
     */
    public function failed_auth_logs_ip_address(): void
    {
        Log::shouldReceive('warning')
            ->atLeast()->once()
            ->withArgs(function ($message, $context) {
                // Should include IP in context
                return isset($context['ip']) || 
                       isset($context['IP']) || 
                       isset($context['client_ip']) ||
                       isset($context['remote_addr']);
            });
        
        $this->postJson('/api/deploy/migrate', ['key' => 'invalid-key']);
    }

    /**
     * @test
     */
    public function failed_auth_logs_user_agent(): void
    {
        Log::shouldReceive('warning')
            ->atLeast()->once()
            ->withArgs(function ($message, $context) {
                // Should include user agent in context
                return isset($context['user_agent']) || 
                       isset($context['userAgent']) ||
                       isset($context['agent']);
            });
        
        $this->postJson('/api/deploy/migrate', ['key' => 'invalid-key'], [
            'User-Agent' => 'TestAgent/1.0',
        ]);
    }

    /**
     * @test
     */
    public function failed_auth_logs_timestamp(): void
    {
        Log::shouldReceive('warning')
            ->atLeast()->once()
            ->withArgs(function ($message, $context) {
                // Timestamp is usually auto-added by Laravel
                return true;
            });
        
        $this->postJson('/api/deploy/migrate', ['key' => 'invalid-key']);
    }

    /**
     * @test
     */
    public function successful_auth_is_not_logged_at_warning_level(): void
    {
        $validKey = env('DEPLOY_KEY', 'test-deploy-key');
        
        Log::shouldReceive('warning')
            ->never();
        
        // This might fail migrations in test, but auth should succeed
        $this->postJson('/api/deploy/migrate', ['key' => $validKey]);
    }
}
```

## Edge Cases

| Case | Expected Behavior |
|------|-------------------|
| Empty deploy key in request | 403 Forbidden, logged |
| Null deploy key in request | 403 Forbidden, logged |
| Very long deploy key (buffer overflow attempt) | 403 Forbidden, logged, no crash |
| Deploy key with special characters | 403 Forbidden if invalid, handled safely |
| Concurrent requests from same IP | Rate limit applies correctly |
| IPv6 addresses | Rate limit works with IPv6 |
| Rate limit reset after window | New requests allowed |
| Deploy during maintenance mode | Handled gracefully |
