## Notes for Developer

### Implementation Requirements

1. **Create `frontend/app/audit/[hash]/page.tsx`:**
   ```typescript
   import { notFound } from 'next/navigation';
   import { Metadata } from 'next';
   import PublicAuditPage from '@/components/audit/PublicAuditPage';

   export const revalidate = 3600; // 1 hour ISR

   interface PageProps {
     params: Promise<{ hash: string }>;
   }

   async function getAuditData(hash: string) {
     const res = await fetch(
       `${process.env.NEXT_PUBLIC_API_URL}/api/v1/public-audit/${hash}`,
       { next: { revalidate: 3600 } }
     );

     if (!res.ok) {
       return null;
     }

     const data = await res.json();
     return data.success ? data.data : null;
   }

   export async function generateMetadata({ params }: PageProps): Promise<Metadata> {
     const { hash } = await params;
     const audit = await getAuditData(hash);

     if (!audit) {
       return {
         title: 'Audit Not Found',
         robots: 'noindex',
       };
     }

     return {
       title: `${audit.siteUrl} SEO Audit Report | SEOIndicator`,
       description: `SEO audit for ${audit.siteUrl} - Score: ${audit.overallScore}/100. View technical SEO analysis and recommendations.`,
       robots: 'index, follow',
       alternates: {
         canonical: `https://seoindicator.com/audit/${hash}/`,
       },
       openGraph: {
         title: `${audit.siteUrl} SEO Score: ${audit.overallScore}/100`,
         description: `SEO audit results for ${audit.siteUrl}`,
         url: `https://seoindicator.com/audit/${hash}/`,
         type: 'website',
         images: [
           {
             url: `https://seoindicator.com/badge/${hash}/og.png`,
             width: 1200,
             height: 630,
             alt: `SEO Score: ${audit.overallScore}/100`,
           },
         ],
       },
       twitter: {
         card: 'summary_large_image',
         title: `${audit.siteUrl} SEO Score: ${audit.overallScore}/100`,
         description: `SEO audit results for ${audit.siteUrl}`,
         images: [`https://seoindicator.com/badge/${hash}/og.png`],
       },
     };
   }

   export default async function Page({ params }: PageProps) {
     const { hash } = await params;
     const audit = await getAuditData(hash);

     if (!audit) {
       notFound();
     }

     return <PublicAuditPage audit={audit} />;
   }
   ```

### Key Points

- **ISR Revalidation**: 3600 seconds (1 hour) for balance between freshness and performance
- **Hash Format**: 32-character alphanumeric string (validated by API)
- **No Authentication**: Public pages accessible without login
- **SEO Optimized**: Full metadata including OG tags and canonical URLs
- **Graceful 404**: User-friendly not found page with CTAs
- **Server-Side Rendering**: Data fetched on server for SEO

### Security Considerations

- Hash is URL-safe (alphanumeric only)
- No sensitive data in public responses
- Private audits return 404 (not 403 to prevent enumeration)
- Rate limiting handled at API level
