# TESTS: SPEC-SEC-2026-02-21-1 — Remove Hardcoded Secrets

## Task
Remove all hardcoded secrets (APP_KEY, DB password, OAuth secrets) from tracked files. Update `.env` and `.env.example` with placeholders only.

## Test Categories

### 1. Unit Tests

#### 1.1 Environment Variable Loading Test
**File:** `tests/Unit/Security/EnvLoadingTest.php`

```php
<?php

namespace Tests\Unit\Security;

use Tests\TestCase;

class EnvLoadingTest extends TestCase
{
    /**
     * Verify APP_KEY is loaded from environment, not hardcoded.
     */
    public function test_app_key_loaded_from_environment(): void
    {
        $appKey = config('app.key');
        
        // Should not be empty
        $this->assertNotEmpty($appKey, 'APP_KEY should be loaded from environment');
        
        // Should start with 'base64:'
        $this->assertStringStartsWith('base64:', $appKey, 'APP_KEY should be base64 encoded');
        
        // Should not be the old hardcoded key
        $this->assertNotEquals(
            '${{ secrets.LARAVEL_APP_KEY }}=',
            $appKey,
            'APP_KEY should not be the old hardcoded value'
        );
    }

    /**
     * Verify database credentials are loaded from environment.
     */
    public function test_db_credentials_from_environment(): void
    {
        $dbPassword = config('database.connections.mysql.password');
        
        // Should not contain hardcoded passwords
        $this->assertNotEquals('hardcoded_password', $dbPassword);
        
        // Should be loaded from env (can be empty for local sqlite)
        $this->assertTrue(
            $dbPassword === env('DB_PASSWORD') || $dbPassword === '',
            'DB_PASSWORD should come from environment'
        );
    }

    /**
     * Verify OAuth credentials are loaded from environment.
     */
    public function test_oauth_credentials_from_environment(): void
    {
        $clientId = config('services.google.client_id');
        $clientSecret = config('services.google.client_secret');
        
        // Should not be empty in production
        if (app()->environment('production')) {
            $this->assertNotEmpty($clientId, 'GOOGLE_CLIENT_ID should be set');
            $this->assertNotEmpty($clientSecret, 'GOOGLE_CLIENT_SECRET should be set');
        }
        
        // Should not be hardcoded values
        $this->assertStringNotContainsString('hardcoded', $clientId ?? '');
        $this->assertStringNotContainsString('hardcoded', $clientSecret ?? '');
    }
}
```

### 2. Security Tests

#### 2.1 No Secrets in Tracked Files Test
**File:** `tests/Security/NoSecretsInTrackedFilesTest.php`

```php
<?php

namespace Tests\Security;

use Tests\TestCase;
use Illuminate\Support\Facades\Process;

class NoSecretsInTrackedFilesTest extends TestCase
{
    /**
     * @test
     * @dataProvider secretPatternsProvider
     */
    public function no_secrets_in_tracked_files(string $pattern, string $description): void
    {
        $result = Process::run('git grep -i "' . $pattern . '" -- ":!.env" ":!.env.*" ":!vendor/*"');
        
        $this->assertEquals(
            1, // Exit code 1 means no matches found
            $result->exitCode(),
            "Found {$description} in tracked files: " . $result->output()
        );
    }

    public static function secretPatternsProvider(): array
    {
        return [
            'APP_KEY hardcoded' => ['base64:xjet1954qKYg8lAySf9OZAiUdVpB', 'hardcoded APP_KEY'],
            'DB_PASSWORD hardcoded' => ['DB_PASSWORD=\'[^\']{8,}\'', 'hardcoded DB_PASSWORD'],
            'OAuth secret pattern' => ['client_secret.*=.*[a-zA-Z0-9]{20,}', 'hardcoded OAuth secret'],
            'API key pattern' => ['api_key.*=.*[a-zA-Z0-9]{20,}', 'hardcoded API key'],
        ];
    }

    /**
     * @test
     */
    public function env_file_is_gitignored(): void
    {
        $gitignore = file_get_contents(base_path('.gitignore'));
        
        $this->assertStringContainsString('.env', $gitignore, '.env should be in .gitignore');
        $this->assertStringNotContainsString('!.env', $gitignore, '.env should not be negated in .gitignore');
    }

    /**
     * @test
     */
    public function env_example_has_placeholders_only(): void
    {
        $envExamplePath = base_path('.env.example');
        
        if (!file_exists($envExamplePath)) {
            $this->markTestSkipped('.env.example not found');
        }
        
        $content = file_get_contents($envExamplePath);
        
        // Should not contain real secrets
        $this->assertStringNotContainsString('base64:', $content, '.env.example should not have APP_KEY');
        $this->assertStringNotContainsString('password=', $content, '.env.example should not have passwords');
        
        // Should have empty or placeholder values
        $this->assertMatchesRegularExpression('/APP_KEY=$/', $content, 'APP_KEY should be empty in .env.example');
    }

    /**
     * @test
     */
    public function deploy_yml_uses_github_secrets(): void
    {
        $deployPath = base_path('../.github/workflows/deploy.yml');
        
        if (!file_exists($deployPath)) {
            $this->markTestSkipped('deploy.yml not found');
        }
        
        $content = file_get_contents($deployPath);
        
        // Should use GitHub Secrets syntax
        $this->assertStringContainsString('${{ secrets.', $content, 'deploy.yml should use GitHub Secrets');
        
        // Should not have hardcoded APP_KEY
        $this->assertStringNotContainsString(
            'APP_KEY=base64:',
            $content,
            'deploy.yml should not have hardcoded APP_KEY'
        );
        
        // Should reference LARAVEL_APP_KEY secret
        $this->assertStringContainsString(
            'LARAVEL_APP_KEY',
            $content,
            'deploy.yml should reference LARAVEL_APP_KEY secret'
        );
    }
}
```

### 3. Integration Tests

#### 3.1 Config Loading Integration Test
**File:** `tests/Integration/Security/ConfigLoadingIntegrationTest.php`

```php
<?php

namespace Tests\Integration\Security;

use Tests\TestCase;

class ConfigLoadingIntegrationTest extends TestCase
{
    /**
     * @test
     */
    public function all_required_env_vars_are_set(): void
    {
        $requiredVars = [
            'APP_KEY',
            'DB_CONNECTION',
            'DB_HOST',
            'DB_DATABASE',
            'DB_USERNAME',
        ];

        foreach ($requiredVars as $var) {
            $value = env($var);
            $this->assertNotEmpty(
                $value,
                "Required environment variable {$var} is not set"
            );
        }
    }

    /**
     * @test
     */
    public function sensitive_values_are_not_logged(): void
    {
        $logContent = file_get_contents(storage_path('logs/laravel.log'));
        
        $sensitivePatterns = [
            'base64:xjet1954qKYg8lAySf9OZAiUdVpB',
            'DB_PASSWORD',
            'GOOGLE_CLIENT_SECRET',
        ];

        foreach ($sensitivePatterns as $pattern) {
            // Only check if the pattern appears in a value context
            $this->assertStringNotContainsString(
                $pattern,
                $logContent,
                "Sensitive value should not appear in logs"
            );
        }
    }
}
```
