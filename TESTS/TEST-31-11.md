## Model Definition Reference

### SEOAnalysis Model Updates

```php
class SEOAnalysis extends Model
{
    protected $fillable = [
        // Existing fields...
        'ai_model',
        'prompt_tokens',
        'response_tokens',
        'total_tokens',
        'truncation_applied',
        'overall_summary',
        'category_scores',
        'ai_request_id',
    ];

    protected $casts = [
        // Existing casts...
        'prompt_tokens' => 'integer',
        'response_tokens' => 'integer',
        'total_tokens' => 'integer',
        'truncation_applied' => 'array',
        'category_scores' => 'array',
    ];

    public function aiRequest(): HasOne
    {
        return $this->hasOne(AIAnalysisRequest::class, 'analysis_id');
    }

    public function wasUnifiedAnalysis(): bool
    {
        return !is_null($this->ai_model);
    }

    public function getCategoryScore(string $category): ?int
    {
        return $this->category_scores[$category] ?? null;
    }

    public function getTruncationStrategies(): array
    {
        return $this->truncation_applied ?? [];
    }

    public function wasTruncated(): bool
    {
        return !empty($this->truncation_applied);
    }

    public function getTokenUsage(): array
    {
        return [
            'prompt' => $this->prompt_tokens,
            'response' => $this->response_tokens,
            'total' => $this->total_tokens,
        ];
    }

    public function getAiModelDisplayName(): string
    {
        if (!$this->ai_model) {
            return 'Unknown';
        }

        return match (true) {
            str_contains($this->ai_model, 'gemini') => 'Gemini 2.0 Flash',
            str_contains($this->ai_model, 'llama') => 'Llama 3 8B',
            str_contains($this->ai_model, 'mistral') => 'Mistral 7B',
            str_contains($this->ai_model, 'qwen') => 'Qwen 2 7B',
            default => $this->ai_model,
        };
    }
}
```

### AIAnalysisRequest Model

```php
class AIAnalysisRequest extends Model
{
    protected $fillable = [
        'analysis_id',
        'model',
        'prompt_tokens',
        'response_tokens',
        'total_tokens',
        'latency_ms',
        'truncation_strategies',
        'request_hash',
        'status',
        'error_message',
    ];

    protected $casts = [
        'prompt_tokens' => 'integer',
        'response_tokens' => 'integer',
        'total_tokens' => 'integer',
        'latency_ms' => 'integer',
        'truncation_strategies' => 'array',
    ];

    protected $attributes = [
        'status' => 'pending',
    ];

    public function analysis(): BelongsTo
    {
        return $this->belongsTo(SEOAnalysis::class, 'analysis_id');
    }

    public function wasSuccessful(): bool
    {
        return $this->status === 'completed';
    }

    public function wasTruncated(): bool
    {
        return !empty($this->truncation_strategies);
    }

    public function getLatencySeconds(): float
    {
        return $this->latency_ms ? $this->latency_ms / 1000 : 0.0;
    }

    public function scopeSuccessful(Builder $query): Builder
    {
        return $query->where('status', 'completed');
    }

    public function scopeFailed(Builder $query): Builder
    {
        return $query->where('status', 'failed');
    }
}
```
