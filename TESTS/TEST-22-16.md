```php
    Queue::assertPushedOn('crawler-low', CrawlWorker::class);
}
```

### Delayed Dispatch Tests

```php
/**
 * Test delayed dispatch (scheduled for later)
 */
public function test_delayed_dispatch(): void
{
    Queue::fake();
    
    $jobId = 'crawl-delayed';
    $delay = now()->addMinutes(5);
    
    CrawlWorker::dispatch($jobId)->delay($delay);
    
    Queue::assertPushed(CrawlWorker::class, function ($job) use ($delay) {
        return $job->delay == $delay;
    });
}
```
