## Mock API Handlers (MSW)

```typescript
// frontend/__tests__/mocks/handlers/aiVisibilityAlerts.ts
import { rest } from 'msw';

export const alertHandlers = [
  // GET /api/v1/ai-visibility/alerts
  rest.get('/api/v1/ai-visibility/alerts', (req, res, ctx) => {
    const page = parseInt(req.url.searchParams.get('page') || '1');
    const perPage = parseInt(req.url.searchParams.get('per_page') || '15');
    const projectId = req.url.searchParams.get('project_id');
    const isRead = req.url.searchParams.get('is_read');
    const alertType = req.url.searchParams.get('alert_type');

    let filtered = [...mockAlerts];
    if (projectId) filtered = filtered.filter(a => a.project_id === projectId);
    if (isRead !== null) filtered = filtered.filter(a => a.is_read === (isRead === 'true'));
    if (alertType) filtered = filtered.filter(a => a.alert_type === alertType);

    const start = (page - 1) * perPage;
    const paginated = filtered.slice(start, start + perPage);

    return res(ctx.json({
      data: paginated,
      meta: {
        current_page: page,
        per_page: perPage,
        total: filtered.length,
        unread_count: filtered.filter(a => !a.is_read).length,
        last_page: Math.ceil(filtered.length / perPage),
      },
    }));
  }),

  // GET /api/v1/ai-visibility/alerts/:id
  rest.get('/api/v1/ai-visibility/alerts/:id', (req, res, ctx) => {
```
