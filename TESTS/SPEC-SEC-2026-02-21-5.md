#### 1.1 No Secrets in Git History Test
**File:** `tests/Security/GitHistorySecretsTest.php`

```php
<?php

namespace Tests\Security;

use Tests\TestCase;
use Illuminate\Support\Facades\Process;

class GitHistorySecretsTest extends TestCase
{
    /**
     * @test
     * @dataProvider secretPatternsProvider
     */
    public function no_secrets_in_git_history(string $pattern, string $description): void
    {
        // Search entire git history for secret patterns
        $result = Process::path(base_path('../'))
            ->run("git log -p --all -S '{$pattern}' --pretty=format:'%H' | head -n 1");
        
        $output = trim($result->output());
        
        $this->assertEmpty(
            $output,
            "Found {$description} in git history"
        );
    }

    public static function secretPatternsProvider(): array
    {
        return [
            'Old APP_KEY' => ['xjet1954qKYg8lAySf9OZAiUdVpB+CcjtNXkJrbEidA=', 'old APP_KEY'],
            'Base64 APP_KEY pattern' => ['base64:xjet1954qKYg8lAySf9OZAiUdVpB', 'APP_KEY fragment'],
            'Hardcoded DB password in deploy.yml' => ["DB_PASSWORD='[^']{8,}'", 'hardcoded DB password'],
            'Google client secret pattern' => ['GOCSPX-', 'Google OAuth secret prefix'],
        ];
    }

    /**
     * @test
     */
    public function no_secrets_in_any_commit(): void
    {
        // Get all commit hashes
        $result = Process::path(base_path('../'))
            ->run('git rev-list --all');
        
        $commits = array_filter(explode("\n", trim($result->output())));
        
        $secretPatterns = [
            'xjet1954qKYg8lAySf9OZAiUdVpB',
            'APP_KEY=base64:xjet',
        ];
        
        foreach (array_slice($commits, 0, 100) as $commit) {
            // Skip if empty
            if (empty($commit)) continue;
            
            $showResult = Process::path(base_path('../'))
                ->run("git show {$commit} 2>/dev/null | head -n 500");
            
            foreach ($secretPatterns as $pattern) {
                $this->assertStringNotContainsString(
                    $pattern,
                    $showResult->output(),
                    "Found secret pattern '{$pattern}' in commit {$commit}"
                );
            }
        }
    }

    /**
     * @test
     */
    public function deploy_yml_history_is_clean(): void
    {
        // Check history of deploy.yml specifically
        $result = Process::path(base_path('../'))
            ->run('git log -p --all -- .github/workflows/deploy.yml 2>/dev/null | head -n 1000');
        
        $output = $result->output();
        
        $this->assertStringNotContainsString(
            'APP_KEY=base64:xjet',
            $output,
            'Old hardcoded APP_KEY should not appear in deploy.yml history'
        );
        
        $this->assertStringNotContainsString(
            'xjet1954qKYg8lAySf9OZAiUdVpB+CcjtNXkJrbEidA=',
            $output,
            'Old APP_KEY value should not appear in deploy.yml history'
        );
    }

    /**
     * @test
     */
    public function no_standalone_env_file_in_history(): void
    {
        // Check if .env file was ever committed
        $result = Process::path(base_path('../'))
            ->run('git log --all --full-history -- .env 2>/dev/null');
        
        $output = trim($result->output());
        
        $this->assertEmpty(
            $output,
            '.env file should never have been committed to git history'
        );
    }

    /**
     * @test
     */
    public function no_backup_files_with_secrets(): void
    {
        // Check for backup files that might contain secrets
        $patterns = [
            '*.env.backup',
            '*.env.bak',
            '*.env.old',
            '.env.*.backup',
        ];
        
        $patternStr = implode(' ', $patterns);
        
        $result = Process::path(base_path('../'))
            ->run("git log --all --full-history -- {$patternStr} 2>/dev/null");
        
        // Should be empty (no backup files ever committed)
        $this->assertEmpty(trim($result->output()));
    }
}
```

#### 3.1 Git History Purge Validation Test
**File:** `tests/Security/GitHistoryPurgeValidationTest.php`

```php
<?php

namespace Tests\Security;

use Tests\TestCase;
use Illuminate\Support\Facades\Process;

class GitHistoryPurgeValidationTest extends TestCase
{
    /**
     * @test
     */
    public function git_filter_repo_or_bfg_is_available(): void
    {
        // Check if git-filter-repo is available
        $filterRepoResult = Process::run('git filter-repo --version 2>/dev/null');
        
        if ($filterRepoResult->successful()) {
            $this->assertTrue(true, 'git filter-repo is available');
            return;
        }
        
        // Check if BFG is available
        $bfgResult = Process::run('bfg --version 2>/dev/null');
        
        if ($bfgResult->successful()) {
            $this->assertTrue(true, 'BFG Repo-Cleaner is available');
            return;
        }
        
        $this->markTestSkipped(
            'Neither git filter-repo nor BFG is installed. ' .
            'Install with: pip install git-filter-repo OR download BFG from https://rtyley.github.io/bfg-repo-cleaner/'
        );
    }

    /**
     * @test
     */
    public function secrets_file_exists_for_purge(): void
    {
        // Document expected secrets file for git history purge
        $secretsFilePath = base_path('../secrets-to-purge.txt');
        
        $expectedContent = <<<'TXT'
# List of secrets to purge from git history
# One per line, exact match

xjet1954qKYg8lAySf9OZAiUdVpB+CcjtNXkJrbEidA=
APP_KEY=${{ secrets.LARAVEL_APP_KEY }}=
TXT;

        $this->assertTrue(true, "Expected secrets file content:\n{$expectedContent}");
    }

    /**
     * @test
     */
    public function git_grep_finds_no_secrets(): void
    {
        $result = Process::path(base_path('../'))
            ->run('git grep -i "xjet1954qKYg8lAySf9OZAiUdVpB" 2>/dev/null');
        
        // Exit code 1 means no matches (what we want)
        // Exit code 0 means matches found (bad)
        $this->assertEquals(
            1,
            $result->exitCode(),
            'git grep should find no secrets: ' . $result->output()
        );
    }
}
```
