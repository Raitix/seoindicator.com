#### 1.1 APP_KEY Generation Test
**File:** `tests/Unit/Security/AppKeyRotationTest.php`

```php
<?php

namespace Tests\Unit\Security;

use Tests\TestCase;
use Illuminate\Support\Facades\Artisan;

class AppKeyRotationTest extends TestCase
{
    /**
     * @test
     */
    public function app_key_is_valid_format(): void
    {
        $appKey = config('app.key');
        
        // Should start with 'base64:'
        $this->assertStringStartsWith('base64:', $appKey);
        
        // Should be 44 characters after 'base64:' (32 bytes encoded)
        $keyPart = substr($appKey, 7);
        $this->assertEquals(44, strlen($keyPart), 'APP_KEY should be 32 bytes (44 base64 chars)');
        
        // Should be valid base64
        $decoded = base64_decode($keyPart);
        $this->assertEquals(32, strlen($decoded), 'Decoded APP_KEY should be 32 bytes');
    }

    /**
     * @test
     */
    public function app_key_is_not_old_hardcoded_value(): void
    {
        $appKey = config('app.key');
        
        $this->assertNotEquals(
            '${{ secrets.LARAVEL_APP_KEY }}=',
            $appKey,
            'APP_KEY must be rotated from the compromised value'
        );
    }

    /**
     * @test
     */
    public function app_key_is_used_for_encryption(): void
    {
        $encrypted = encrypt('test-value');
        
        $this->assertNotEquals('test-value', $encrypted);
        $this->assertEquals('test-value', decrypt($encrypted));
    }

    /**
     * @test
     */
    public function new_key_generates_different_ciphertext(): void
    {
        $value = 'sensitive-data';
        
        $encrypted1 = encrypt($value);
        $encrypted2 = encrypt($value);
        
        // Each encryption should produce different ciphertext (random IV)
        $this->assertNotEquals($encrypted1, $encrypted2);
    }
}
```

### 2. Integration Tests

#### 2.1 Deploy Workflow Secret Integration Test
**File:** `tests/Integration/Security/DeploySecretsIntegrationTest.php`

```php
<?php

namespace Tests\Integration\Security;

use Tests\TestCase;

class DeploySecretsIntegrationTest extends TestCase
{
    /**
     * @test
     */
    public function deploy_yml_uses_secrets_for_app_key(): void
    {
        $deployPath = base_path('../.github/workflows/deploy.yml');
        
        if (!file_exists($deployPath)) {
            $this->markTestSkipped('deploy.yml not found');
        }
        
        $content = file_get_contents($deployPath);
        
        // Should use GitHub Secret reference
        $this->assertMatchesRegularExpression(
            '/APP_KEY=\$\{\{ secrets\.LARAVEL_APP_KEY \}\}/',
            $content,
            'deploy.yml should use LARAVEL_APP_KEY secret'
        );
    }

    /**
     * @test
     */
    public function deploy_yml_uses_secrets_for_db_password(): void
    {
        $deployPath = base_path('../.github/workflows/deploy.yml');
        
        if (!file_exists($deployPath)) {
            $this->markTestSkipped('deploy.yml not found');
        }
        
        $content = file_get_contents($deployPath);
        
        // Should use GitHub Secret reference for DB_PASSWORD
        $this->assertMatchesRegularExpression(
            '/DB_PASSWORD=.*\$\{\{ secrets\.DB_PASSWORD \}\}/',
            $content,
            'deploy.yml should use DB_PASSWORD secret'
        );
    }

    /**
     * @test
     */
    public function encryption_works_with_current_app_key(): void
    {
        // Test that OAuth tokens can be encrypted/decrypted
        $testToken = 'test-access-token-' . str_repeat('x', 100);
        
        $encrypted = encrypt($testToken);
        $decrypted = decrypt($encrypted);
        
        $this->assertEquals($testToken, $decrypted);
    }

    /**
     * @test
     */
    public function session_encryption_works(): void
    {
        // Store and retrieve session data
        session(['test_secret' => 'sensitive-value']);
        
        $this->assertEquals('sensitive-value', session('test_secret'));
    }
}
```

## Edge Cases

| Case | Expected Behavior |
|------|-------------------|
| APP_KEY rotation mid-session | User session invalidated, forced to re-login |
| Encrypted data in database | Data encrypted with old key becomes unreadable |
| Queue jobs with encrypted payloads | Jobs fail, need to be re-queued |
| Cached config with encrypted values | Cache must be cleared |
| Multiple app instances | All must have same APP_KEY |
