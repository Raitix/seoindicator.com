### 2. Redirect URI Validation Tests

#### 2.1 Redirect URI Security Test
**File:** `tests/Security/OAuthRedirectUriSecurityTest.php`

```php
<?php

namespace Tests\Security;

use Tests\TestCase;
use App\Services\OAuthService;
use RuntimeException;

class OAuthRedirectUriSecurityTest extends TestCase
{
    protected OAuthService $service;

    protected function setUp(): void
    {
        parent::setUp();
        $this->service = app(OAuthService::class);
    }

    /**
     * @test
     */
    public function valid_redirect_uris_are_accepted(): void
    {
        $validUris = $this->service->getValidRedirectUris();
        
        $this->assertNotEmpty($validUris, 'Should have at least one valid redirect URI');
    }

    /**
     * @test
     * @dataProvider maliciousRedirectUriProvider
     */
    public function malicious_redirect_uris_are_rejected(string $maliciousUri, string $description): void
    {
        $this->expectException(RuntimeException::class);
        $this->expectExceptionMessage('not allowed');
        
        $this->service->getAuthorizationUrl($maliciousUri);
    }

    public static function maliciousRedirectUriProvider(): array
    {
        return [
            'Open redirect to external site' => ['https://evil.com/callback', 'external site'],
            'JavaScript protocol' => ['javascript:alert(document.cookie)', 'javascript protocol'],
            'Data URI' => ['data:text/html,<script>alert(1)</script>', 'data URI'],
            'Local file' => ['file:///etc/passwd', 'file protocol'],
            'SSRF to internal' => ['http://localhost/admin', 'SSRF localhost'],
            'SSRF to internal IP' => ['http://127.0.0.1:8080/callback', 'SSRF 127.0.0.1'],
            'SSRF to AWS metadata' => ['http://169.254.169.254/latest/meta-data/', 'SSRF AWS metadata'],
            'Open redirect via parameter' => ['http://localhost:3000/gsc/callback?redirect=evil.com', 'redirect param'],
            'Unicode bypass' => ['http://localhost:3000/gsc/callбack', 'unicode bypass'],
            'Null byte injection' => ["http://localhost:3000/gsc/callback\x00.evil.com", 'null byte'],
        ];
    }

    /**
     * @test
     */
    public function production_redirect_uri_must_be_https(): void
    {
        if (!app()->environment('production')) {
            $this->markTestSkipped('Only applies to production');
        }
        
        $redirectUri = config('services.google.redirect_uri');
        
        $this->assertStringStartsWith(
            'https://',
            $redirectUri,
            'Production redirect URI must use HTTPS'
        );
    }

    /**
     * @test
     */
    public function redirect_uri_must_match_configured_list(): void
    {
        $service = $this->service;
        $validUris = $service->getValidRedirectUris();
        
        // Any URI not in the valid list should throw
        $invalidUri = 'https://not-in-list.com/callback';
        
        $this->assertNotContains($invalidUri, $validUris);
        
        $this->expectException(RuntimeException::class);
        $service->getAuthorizationUrl($invalidUri);
    }
}
```

### 3. Token Encryption Tests

#### 3.1 Token Encryption at Rest Test
**File:** `tests/Integration/Security/OAuthTokenEncryptionTest.php`

```php
<?php

namespace Tests\Integration\Security;

use Tests\TestCase;
use App\Models\GSCConnection;
use App\Models\User;
use Illuminate\Support\Facades\Crypt;
use Illuminate\Support\Facades\DB;

class OAuthTokenEncryptionTest extends TestCase
{
    /**
     * @test
     */
    public function access_token_is_encrypted_in_database(): void
    {
        $user = User::factory()->create();
        
        $connection = GSCConnection::create([
            'user_id' => $user->id,
            'site_url' => 'https://example.com',
            'property_url' => 'https://example.com/',
            'access_token' => 'test-access-token-12345',
            'refresh_token' => 'test-refresh-token-67890',
            'token_expires_at' => now()->addHour(),
            'status' => 'active',
        ]);
        
        // Retrieve raw value from database
        $rawToken = DB::table('gsc_connections')
            ->where('id', $connection->id)
            ->value('access_token');
        
        // Raw value should NOT be the plaintext token
        $this->assertNotEquals('test-access-token-12345', $rawToken);
        
        // Raw value should be encrypted (base64-like)
        $this->assertMatchesRegularExpression(
            '/^[A-Za-z0-9+\/=]+$/',
            $rawToken,
            'Access token should be encrypted (base64 format)'
        );
        
        // Model accessor should decrypt it
        $this->assertEquals('test-access-token-12345', $connection->access_token);
    }

    /**
     * @test
     */
    public function refresh_token_is_encrypted_in_database(): void
    {
        $user = User::factory()->create();
        
        $connection = GSCConnection::create([
            'user_id' => $user->id,
            'site_url' => 'https://example.com',
            'property_url' => 'https://example.com/',
            'access_token' => 'test-access',
            'refresh_token' => 'test-refresh-token-secret',
            'token_expires_at' => now()->addHour(),
            'status' => 'active',
        ]);
        
        $rawToken = DB::table('gsc_connections')
            ->where('id', $connection->id)
            ->value('refresh_token');
        
        // Raw value should be encrypted
        $this->assertNotEquals('test-refresh-token-secret', $rawToken);
        
        // Model accessor should decrypt it
        $this->assertEquals('test-refresh-token-secret', $connection->refresh_token);
    }

    /**
     * @test
     */
    public function tokens_are_hidden_from_serialization(): void
    {
        $user = User::factory()->create();
        
        $connection = GSCConnection::create([
            'user_id' => $user->id,
            'site_url' => 'https://example.com',
            'property_url' => 'https://example.com/',
            'access_token' => 'secret-access-token',
            'refresh_token' => 'secret-refresh-token',
            'token_expires_at' => now()->addHour(),
            'status' => 'active',
        ]);
        
        $json = $connection->toJson();
        $array = json_decode($json, true);
        
        $this->assertArrayNotHasKey('access_token', $array);
        $this->assertArrayNotHasKey('refresh_token', $array);
        
        // Should not contain the actual tokens anywhere
        $this->assertStringNotContainsString('secret-access-token', $json);
        $this->assertStringNotContainsString('secret-refresh-token', $json);
    }

    /**
     * @test
     */
    public function encryption_uses_app_key(): void
    {
        // Verify that encryption is working
        $encrypted = encrypt('test-value');
        
        // Should be decryptable
        $this->assertEquals('test-value', decrypt($encrypted));
        
        // Should be different from plaintext
        $this->assertNotEquals('test-value', $encrypted);
    }
}
```

### 4. Integration Tests

#### 4.1 OAuth Service Integration Test
**File:** `tests/Integration/Security/OAuthServiceIntegrationTest.php`

```php
<?php

namespace Tests\Integration\Security;

use Tests\TestCase;
use App\Services\OAuthService;
use App\Models\User;

class OAuthServiceIntegrationTest extends TestCase
{
    /**
     * @test
     */
    public function authorization_url_contains_required_parameters(): void
    {
        $service = app(OAuthService::class);
        $redirectUri = 'http://localhost:3000/gsc/callback';
        
        $url = $service->getAuthorizationUrl($redirectUri);
        
        $this->assertStringContainsString('client_id=', $url);
        $this->assertStringContainsString('redirect_uri=', urlencode($redirectUri));
        $this->assertStringContainsString('response_type=code', $url);
        $this->assertStringContainsString('scope=', $url);
        $this->assertStringContainsString('state=', $url);
        $this->assertStringContainsString('access_type=offline', $url);
    }

    /**
     * @test
     */
    public function authorization_url_uses_correct_scopes(): void
    {
        $service = app(OAuthService::class);
        $redirectUri = 'http://localhost:3000/gsc/callback';
        
        $url = $service->getAuthorizationUrl($redirectUri);
        
        // Should request webmasters.readonly scope
        $this->assertStringContainsString(
            urlencode('https://www.googleapis.com/auth/webmasters.readonly'),
            $url
        );
    }

    /**
     * @test
     */
    public function state_parameter_is_included_for_csrf_protection(): void
    {
        $service = app(OAuthService::class);
        $redirectUri = 'http://localhost:3000/gsc/callback';
        
        $url1 = $service->getAuthorizationUrl($redirectUri);
        $url2 = $service->getAuthorizationUrl($redirectUri, 'custom-state'));
        
        // Should have state parameter
        $this->assertStringContainsString('state=', $url1);
        
        // Custom state should be used
        $this->assertStringContainsString('state=custom-state', $url2);
    }
}
```

## Edge Cases

| Case | Expected Behavior |
|------|-------------------|
| Empty client ID | RuntimeException thrown |
| Empty client secret | RuntimeException thrown |
| Expired token | Token refresh attempted |
| Invalid refresh token | Connection marked as expired |
| Token refresh during race condition | Only one refresh succeeds |
| Redirect URI with port | Must match configured URI exactly |
| Redirect URI with query params | Must match configured URI exactly |
| Non-ASCII in redirect URI | Must be properly encoded |
