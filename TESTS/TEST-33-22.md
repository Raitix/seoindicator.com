## Test Component Mock

```typescript
/**
 * Mock VisibilitySettings component for TDD
 */
export function VisibilitySettings({
  auditId,
  auditHash,
  isPublic: initialIsPublic,
  onPublish,
  onUnpublish,
  className,
}: VisibilitySettingsProps) {
  const [isPublic, setIsPublic] = React.useState(initialIsPublic);
  const [isLoading, setIsLoading] = React.useState(false);
  const [showConfirm, setShowConfirm] = React.useState(false);
  const [copied, setCopied] = React.useState(false);

  const publicUrl = typeof window !== 'undefined'
    ? `${window.location.origin}/audit/${auditHash}`
    : `/audit/${auditHash}`;

  const handleToggle = async () => {
    if (isPublic) {
      setShowConfirm(true);
    } else {
      setIsLoading(true);
      try {
        await onPublish();
        setIsPublic(true);
        toast.success('Audit published successfully');
      } catch (error) {
        toast.error('Failed to publish audit');
      } finally {
        setIsLoading(false);
      }
    }
  };

  const handleUnpublish = async () => {
    setIsLoading(true);
    try {
      await onUnpublish();
      setIsPublic(false);
      setShowConfirm(false);
      toast.success('Audit is now private');
    } catch (error) {
      toast.error('Failed to unpublish audit');
    } finally {
      setIsLoading(false);
    }
  };

  const handleCopy = async () => {
    await navigator.clipboard.writeText(publicUrl);
    setCopied(true);
    setTimeout(() => setCopied(false), 2000);
  };

  return (
    <div className={cn('space-y-4', className)} data-testid="visibility-settings">
      {/* Implementation... */}
    </div>
  );
}
```
