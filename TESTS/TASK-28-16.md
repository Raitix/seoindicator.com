const MockIndicatorCard: React.FC<{
  title: string;
  value: string | number;
  status: 'good' | 'warning' | 'critical' | 'neutral';
  onClick?: () => void;
  disabled?: boolean;
  isLoading?: boolean;
  error?: Error | null;

  return (
    <div
      role="button"
      tabIndex={disabled ? -1 : 0}
      aria-label={`${title}: ${value}. Status: ${status}. Click to view details.`}
      aria-disabled={disabled}
      aria-busy={isLoading}
      onClick={onClick}
      onKeyDown={(e) => {
        if (e.key === 'Enter' || e.key === ' ') {
          e.preventDefault();
          onClick?.();
        }
      }}
      data-testid="indicator-card"
    >
      <span>{title}</span>
      <span>{value}</span>
    </div>
  );
};

## Notes for Developer

### Critical ARIA Patterns

1. **Interactive Cards:**
   - Must have `role="button"` and `tabIndex={0}`
   - Must respond to Enter AND Space keys
   - Include full context in `aria-label`

2. **Modals/Sheets:**
   - Must have `aria-modal="true"`
   - Must trap focus within
   - Must return focus on close

3. **Tabs:**
   - Use `role="tablist"`, `role="tab"`, `role="tabpanel"`
   - Arrow keys navigate, Enter/Space activates
   - Home/End jump to first/last

4. **Dropdowns:**
   - Use `role="menu"`, `role="menuitem"`
   - Escape closes
   - Arrow keys navigate

### Testing Checklist for Developer
- [ ] All interactive elements reachable via Tab
- [ ] Tab order is logical (follows visual order)
- [ ] Focus indicator is always visible
- [ ] Escape closes modals/sheets
- [ ] Enter/Space activate buttons
- [ ] Arrow keys work in menus/tabs
- [ ] Screen reader announces all content correctly
- [ ] Color contrast meets WCAG AA (4.5:1 text, 3:1 UI)

- **jest-axe:** Jest integration for axe
