## Test Patterns

### Component Mocking
```typescript
// Mock child components for isolation
jest.mock('@/components/ui/some-component', () => ({
  SomeComponent: ({ children }: { children: React.ReactNode }) => (
    <div data-testid="mock-some-component">{children}</div>
  ),
}));
```

### Data Mocking
```typescript
// Use type guards for mock validation
import { isCompetitorData, isFeatureCategory } from '@/lib/compare/types';

const mockFeatures: FeatureCategory[] = [
  {
    name: 'Rank Tracking',
    features: [
      { name: 'Daily Updates', seoindicator: true, competitor: true },
      { name: 'Mobile Tracking', seoindicator: true, competitor: false },
    ],
  },
];
```

### Accessibility Testing
```typescript
// Use jest-axe for a11y testing
import { axe, toHaveNoViolations } from 'jest-axe';

expect.extend(toHaveNoViolations);

it('has no accessibility violations', async () => {
  const { container } = render(<Component />);
  const results = await axe(container);
  expect(results).toHaveNoViolations();
});
```
