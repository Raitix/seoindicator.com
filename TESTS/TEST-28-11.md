## Component Interfaces

### LinearProgressProps

```typescript
interface LinearProgressProps {
  value?: number;  // 0-100, undefined for indeterminate
  color?: 'primary' | 'secondary' | 'success' | 'warning' | 'error';
  className?: string;
}
```

```typescript
interface CircularProgressProps {
  value?: number;  // 0-100, undefined for indeterminate
  size?: 'sm' | 'md' | 'lg';
  color?: 'primary' | 'secondary' | 'white';
  className?: string;
}
```

### LoadingState
```typescript
interface LoadingState {
  isLoading: boolean;
  setIsLoading: (loading: boolean) => void;
  error: Error | null;
  setError: (error: Error | null) => void;
```

```typescript
  isPending: boolean;
```

## Implementation Notes for Developer

### Loading State Pattern
```tsx
// Page-level loading
const { isLoading, error, retry } = useLoadingState(fetchIndicators);

return (
  <div>
    {isLoading && <LinearProgress />}
    {isLoading && <IndicatorCatalogSkeleton count={8} />}
    {!isLoading && !error && <IndicatorCatalog indicators={data} />}
    {error && <ErrorBoundary error={error} onRetry={retry} />}
  </div>
);
```

### Button Loading Pattern
```tsx
// Button loading state
const [isAnalyzing, setIsAnalyzing] = useState(false);

return (
  <button disabled={isAnalyzing} onClick={handleAnalyze}>
    {isAnalyzing ? (
      <>
        <CircularProgress size="sm" />
        <span>Analyzing...</span>
      </>
    ) : (
      'Analyze'
    )}
  </button>
);
```

### Optimistic Update Pattern
```tsx
// Optimistic update
const { data, optimisticData, isPending, update, rollback } = 
  useOptimisticUpdate(updateKeywordPosition);

const displayValue = optimisticData?.position ?? data?.position;

return (
  <div>
    <span data-optimistic={!!optimisticData}>{displayValue}</span>
    {isPending && <CircularProgress size="sm" />}
    {error && <span>Update failed</span>}
  </div>
);
```

### Error Boundary Pattern
```tsx
// Error boundary with retry
const { error, retry, retryCount } = useLoadingState(fetchData);

if (error) {
  return (
    <div role="alert">
      <h3>Error Loading Content</h3>
      <p>{error.message}</p>
      {retryCount < 3 ? (
        <button onClick={retry}>Try Again</button>
      ) : (
        <button disabled>Max Retries Reached</button>
      )}
    </div>
  );
}
```

### Empty State Pattern
```tsx
// Empty state with CTA
if (!data || data.length === 0) {
  return (
    <div className="empty-state">
      <h3>No keywords found</h3>
      <p>Add your first keyword to start tracking rankings</p>
      <button onClick={handleAddKeyword}>Add Keyword</button>
    </div>
  );
}
```
