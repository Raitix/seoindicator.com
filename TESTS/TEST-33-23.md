### Unlock Utility Functions

```typescript
// frontend/lib/audit/public-audit.ts

const COOKIE_PREFIX = 'audit_unlocked_';
const COOKIE_EXPIRY_DAYS = 7;

/**
 * Check if an audit is unlocked for the current user
 */
export function isUnlocked(hash: string): boolean {
  const cookieName = `${COOKIE_PREFIX}${hash}`;
  const sessionKey = `${COOKIE_PREFIX}${hash}`;
  
  // Check cookie first
  if (document.cookie.includes(`${cookieName}=`)) return true;
  
  // Fallback to sessionStorage
  if (sessionStorage.getItem(sessionKey) === '1') return true;
  
  return false;
}

/**
 * Set unlock state for an audit
 */
export function setUnlockCookie(hash: string): void {
  const cookieName = `${COOKIE_PREFIX}${hash}`;
  const expires = new Date();
  expires.setDate(expires.getDate() + COOKIE_EXPIRY_DAYS);
  
  try {
    document.cookie = `${cookieName}=1; expires=${expires.toUTCString()}; path=/; SameSite=Lax`;
  } catch {
    // Fallback to sessionStorage if cookies are disabled
    sessionStorage.setItem(cookieName, '1');
  }
}

/**
 * Clear unlock state for an audit
 */
export function clearUnlockCookie(hash: string): void {
  const cookieName = `${COOKIE_PREFIX}${hash}`;
  document.cookie = `${cookieName}=; expires=Thu, 01 Jan 1970 00:00:00 UTC; path=/;`;
  sessionStorage.removeItem(cookieName);
}

/**
 * Generate signup redirect URL with pre-filled data
 */
export function getSignupUrl(email: string, auditHash: string): string {
  const params = new URLSearchParams({
    email: email,
    audit: auditHash
  });
  return `/signup?${params.toString()}`;
}
```
