## Notes for Developer

### Implementation Requirements

1. **Create `frontend/lib/audit/json-ld-schema.ts`:**
   ```typescript
   export interface JsonLdSchema {
     '@context': string;
     '@type': string;
     name: string;
     description?: string;
     url?: string;
     publisher?: {
       '@type': string;
       name: string;
       url: string;
       logo?: {
         '@type': string;
         url: string;
       };
     };
     mainEntity?: {
       '@type': string;
       result?: {
         '@type': string;
         ratingValue: string;
         bestRating: string;
         worstRating: string;
       };
     };
     datePublished?: string;
     dateModified?: string;
   }

   export function generateJsonLdSchema(
     audit: PublicAuditData | null,
     siteUrl = 'https://seoindicator.com'
   ): JsonLdSchema | null {
     if (!audit) return null;

     const displayUrl = audit.siteUrl
       .replace(/^https?:\/\//, '')
       .replace(/\/$/, '');

     return {
       '@context': 'https://schema.org',
       '@type': 'WebPage',
       name: `${displayUrl} SEO Audit Report`,
       description: `SEO audit results for ${displayUrl} - Score: ${audit.overallScore}/100`,
       url: `${siteUrl}/audit/${audit.publicHash}/`,
       publisher: {
         '@type': 'Organization',
         name: 'SEOIndicator',
         url: siteUrl,
         logo: {
           '@type': 'ImageObject',
           url: `${siteUrl}/logo.png`,
         },
       },
       mainEntity: {
         '@type': 'Analysis',
         result: {
           '@type': 'Rating',
           ratingValue: String(audit.overallScore),
           bestRating: '100',
           worstRating: '0',
         },
       },
       datePublished: audit.auditedAt,
       dateModified: audit.auditedAt,
     };
   }

   export function isValidJsonLd(schema: unknown): schema is JsonLdSchema {
     if (!schema || typeof schema !== 'object') return false;
     const s = schema as Record<string, unknown>;
     return (
       s['@context'] === 'https://schema.org' &&
       typeof s['@type'] === 'string' &&
       typeof s.name === 'string'
     );
   }
   ```

2. **Usage in Public Audit Page (`frontend/app/audit/[hash]/page.tsx`):**
   ```typescript
   import { generateJsonLdSchema } from '@/lib/audit/json-ld-schema';

   export default async function PublicAuditPage({ params }) {
     const audit = await getAuditData(params.hash);
     const jsonLd = generateJsonLdSchema(audit);

     return (
       <>
         {jsonLd && (
           <script
             type="application/ld+json"
             dangerouslySetInnerHTML={{ __html: JSON.stringify(jsonLd) }}
           />
         )}
         <AuditContent audit={audit} />
       </>
     );
   }
   ```

3. **Alternative with Next.js Script component:**
   ```typescript
   import Script from 'next/script';
   import { generateJsonLdSchema } from '@/lib/audit/json-ld-schema';

   export default async function PublicAuditPage({ params }) {
     const audit = await getAuditData(params.hash);
     const jsonLd = generateJsonLdSchema(audit);

     return (
       <>
         {jsonLd && (
           <Script id="json-ld" type="application/ld+json">
             {JSON.stringify(jsonLd)}
           </Script>
         )}
         <AuditContent audit={audit} />
       </>
     );
   }
   ```

### Key Points

- **Schema Types:** WebPage (root) → Analysis (mainEntity) → Rating (result)
- **Rating Scale:** 0-100 (bestRating=100, worstRating=0)
- **Publisher:** SEOIndicator organization with logo
- **URLs:** All absolute HTTPS URLs
- **Dates:** ISO 8601 format from auditedAt field
- **JSON-LD Type:** application/ld+json in script tag
- **Validation:** Schema must pass schema.org validators and Google Rich Results Test
