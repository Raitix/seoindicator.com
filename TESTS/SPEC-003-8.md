### Recharts Example Structure
```tsx
import { LineChart, Line, XAxis, YAxis, Tooltip, Legend, ResponsiveContainer } from 'recharts';

<ResponsiveContainer width="100%" height={400}>
  <LineChart data={historyData}>
    <XAxis dataKey="date" />
    <YAxis reversed domain={[1, 100]} />
    <Tooltip content={<CustomTooltip />} />
    <Legend />
    <Line type="monotone" dataKey="user" stroke="#2563eb" />
    <Line type="monotone" dataKey="competitor1" stroke="#dc2626" />
    <Line type="monotone" dataKey="competitor2" stroke="#16a34a" />
  </LineChart>
</ResponsiveContainer>
```
